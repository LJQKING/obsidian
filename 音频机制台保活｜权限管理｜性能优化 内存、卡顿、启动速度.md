```kotlin
// 原理：通过AudioManager管理不同的音频流
// Android定义7种音频流：STREAM_VOICE_CALL, STREAM_SYSTEM, STREAM_RING, 
// STREAM_MUSIC, STREAM_ALARM, STREAM_NOTIFICATION, STREAM_DTMF

class NavigationAudioManager(private val context: Context) {
    private val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as AudioManager
    
    // 原理：STREAM_MUSIC是APP最常用，STREAM_NOTIFICATION用于提示音
    fun setNavigationVolume(volume: Int) {

        val maxVolume = audioManager.getStreamMaxVolume(AudioManager.STREAM_MUSIC)
        
        // 设置当前音量（0-maxVolume）
        audioManager.setStreamVolume(
            AudioManager.STREAM_MUSIC,
            volume.coerceIn(0, maxVolume),
            AudioManager.FLAG_SHOW_UI  // 显示音量条UI
        )
    }
    
    // 原理：不同流不同音量，导航语音用STREAM_NOTIFICATION避免被音乐覆盖
    fun setNavigationAlertVolume(volume: Int) {
        val maxNotifVolume = audioManager.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION)
        audioManager.setStreamVolume(
            AudioManager.STREAM_NOTIFICATION,
            volume.coerceIn(0, maxNotifVolume),
            0
        )
    }
    
    // 获取当前音量
    fun getCurrentVolume(): Int {
        return audioManager.getStreamVolume(AudioManager.STREAM_MUSIC)
    }
}
```

### 1.4 音频焦点管理 - 关键机制

**原理**：Android只允许一个APP独占音频焦点播放，防止多个APP音频混乱

```kotlin
class AudioFocusManager(private val context: Context) {
    private val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as AudioManager
    
    // 焦点监听器
    private val audioFocusRequest: AudioFocusRequest by lazy {
        AudioFocusRequest.Builder(AudioManager.AUDIOFOCUS_GAIN)
            .setAudioAttributes(
                AudioAttributes.Builder()
                    .setUsage(AudioAttributes.USAGE_MEDIA)  // 音乐播放
                    .build()
            )
            .setOnAudioFocusChangeListener(audioFocusChangeListener)
            .build()
    }
    
    private val audioFocusChangeListener = AudioManager.OnAudioFocusChangeListener { focusChange ->
        when (focusChange) {
            AudioManager.AUDIOFOCUS_GAIN -> {
                // 获得焦点：恢复播放
                resumePlayback()
            }
            AudioManager.AUDIOFOCUS_LOSS -> {
                // 永久失去焦点：停止播放
                stopPlayback()
            }
            AudioManager.AUDIOFOCUS_LOSS_TRANSIENT -> {
                // 临时失去焦点（接电话）：暂停播放
                pausePlayback()
            }
            AudioManager.AUDIOFOCUS_LOSS_TRANSIENT_CAN_DUCK -> {
                // 临时失去焦点但可降低音量：降低音量继续播放
                lowerVolume()
            }
        }
    }
    
    fun requestAudioFocus(): Boolean {
        return audioManager.requestAudioFocus(audioFocusRequest) == AudioManager.AUDIOFOCUS_REQUEST_GRANTED
    }
    
    fun abandonAudioFocus() {
        audioManager.abandonAudioFocusRequest(audioFocusRequest)
    }
    
    private fun resumePlayback() { /* ... */ }
    private fun stopPlayback() { /* ... */ }
    private fun pausePlayback() { /* ... */ }
    private fun lowerVolume() { /* ... */ }
}
```

### 1.6 MediaPlayer实现 - 具体场景

**场景2：播放导航语音**

```kotlin
class NavigationVoicePlayer(private val context: Context) : MediaPlayer.OnCompletionListener {
    private var mediaPlayer: MediaPlayer? = null
    private val audioFocusManager = AudioFocusManager(context)
    
    // 原理：MediaPlayer状态机管理
    // Idle → Initialized → Prepared → Started → Paused/Playback Completed
    
    fun playNavigationVoice(uriOrPath: String) {
        // 1. 请求音频焦点
        if (!audioFocusManager.requestAudioFocus()) {
            return  // 无法获得焦点，不播放
        }
        
        try {
            // 2. 初始化MediaPlayer
            mediaPlayer?.release()
            mediaPlayer = MediaPlayer().apply {
                // 3. 设置数据源（支持本地文件、网络URI、资源）
                setDataSource(context, Uri.parse(uriOrPath))
                
                // 4. 设置音频属性（原理：定义音频特征供系统路由）
                setAudioAttributes(
                    AudioAttributes.Builder()
                        .setUsage(AudioAttributes.USAGE_NAVIGATION)  // 导航语音特征
                        .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
                        .build()
                )
                
                // 5. 完成回调
                setOnCompletionListener(this@NavigationVoicePlayer)
                
                // 6. 异步准备（不阻塞主线程）
                setOnPreparedListener { player ->
                    player.start()
                }
                prepareAsync()
            }
        } catch (e: IOException) {
            e.printStackTrace()
        }
    }
    
    override fun onCompletion(mp: MediaPlayer?) {
        audioFocusManager.abandonAudioFocus()
        mediaPlayer?.release()
        mediaPlayer = null
    }
    
    fun stop() {
        mediaPlayer?.apply {
            if (isPlaying) stop()
            release()
        }
        mediaPlayer = null
        audioFocusManager.abandonAudioFocus()
    }
}
```

### 1.7 AudioTrack - 底层音频播放

**原理**：直接操作PCM音频数据，用于实时音频处理

```kotlin
class RealTimeAudioProcessor {
    private lateinit var audioTrack: AudioTrack
    private val sampleRate = 44100  // 采样率
    private val channelConfig = AudioFormat.CHANNEL_OUT_MONO  // 单声道
    private val audioFormat = AudioFormat.ENCODING_PCM_16BIT  // 16位
    
    // 原理：计算缓冲区大小 = 采样率 * 时间 * 字节数
    fun initialize() {
        val bufferSize = AudioTrack.getMinBufferSize(
            sampleRate, 
            channelConfig, 
            audioFormat
        )
        
        audioTrack = AudioTrack(
            AudioAttributes.Builder()
                .setUsage(AudioAttributes.USAGE_MEDIA)
                .build(),
            AudioFormat.Builder()
                .setSampleRate(sampleRate)
                .setChannelMask(channelConfig)
                .setEncoding(audioFormat)
                .build(),
            bufferSize,
            AudioTrack.MODE_STREAM,  // 流式播放（支持流式写入）
            AudioManager.AUDIO_SESSION_ID_GENERATE
        )
    }
    
    // 写入PCM音频数据
    fun playAudio(audioData: ByteArray) {
        audioTrack.play()
        audioTrack.write(audioData, 0, audioData.size)
    }
    
    fun stop() {
        audioTrack.stop()
        audioTrack.release()
    }
}
```

## 二、后台保活机制

### 2.1 后台进程分类

**原理**：Android将进程分为不同优先级，OOM杀手根据优先级回收

```
优先级从高到低：
1. Foreground Process（前台进程）- 拥有可见Activity/Service
2. Visible Process（可见进程）- 拥有可见但不焦点的Activity
3. Service Process（服务进程）- 拥有后台Service
4. Cached Process（缓存进程）- 无组件，最容易被杀
```

### 2.2 前台服务 - 最可靠的保活

**场景1：车机导航持续运行**

```kotlin
// AndroidManifest.xml
<uses-permission android:name="android.permission.FOREGROUND_SERVICE" />
<uses-permission android:name="android.permission.FOREGROUND_SERVICE_LOCATION" />

<service
```

```kotlin
class NavigationService : Service() {
    private val notificationId = 1
    
    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        // 创建前台通知
        val notification = createNotification()
        
        // SDK 29+ 必须使用notificationId
        startForeground(notificationId, notification)
        
        // 后台任务
        startNavigation()
        
        // 返回值说明：
        // START_STICKY: 被杀死后，系统会重新启动服务（Intent为null）
        // START_NOT_STICKY: 被杀死后不自动重启
        // START_REDELIVER_INTENT: 被杀死后，会重新传递之前的Intent
        return START_STICKY
    }
    
    private fun createNotification(): Notification {
        // 创建通知频道（Android 8.0+）
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                "navigation_channel",
                "导航服务",
                NotificationManager.IMPORTANCE_LOW  // 低重要性不会打断用户
            )
            val manager = getSystemService(NotificationManager::class.java)
            manager.createNotificationChannel(channel)
        }
        
        return NotificationCompat.Builder(this, "navigation_channel")
            .setContentTitle("导航中")
            .setContentText("正在导航到目的地")
            .setSmallIcon(R.drawable.ic_navigation)
            .setContentIntent(createPendingIntent())
            .setOngoing(true)  // 用户无法滑动删除
            .build()
    }
    
    private fun startNavigation() {
        // 在后台线程执行
        Thread {
            while (isNavigating) {
                updateLocation()
                Thread.sleep(1000)
            }
        }.start()
    }
    
    override fun onDestroy() {
        super.onDestroy()
        // 被杀死时尝试重启
        startService(Intent(this, NavigationService::class.java))
    }
}
```

### 2.6 保活黑科技 - 双进程守护（注意：某些场景会被系统杀死）

**原理**：两个进程互相监听，其中一个被杀时自动重启另一个

```kotlin
// 进程A：主服务
class MainService : Service() {
    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        // 定期检查进程B是否存在
        checkGuardianProcess()
        
        return START_STICKY
    }
    
    private fun checkGuardianProcess() {
        val timer = Timer()
        timer.scheduleAtFixedRate(object : TimerTask() {
            override fun run() {
                if (!isProcessRunning("GuardianService")) {
                    // 进程B被杀，重启它
                    startService(Intent(this@MainService, GuardianService::class.java))
                }
            }
        }, 0, 5000)  // 每5秒检查一次
    }
    
    private fun isProcessRunning(processName: String): Boolean {
        val activityManager = getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager
        return activityManager.runningAppProcesses.any {
            it.processName.endsWith(processName)
        }
    }
}

// 进程B：守护进程（在AndroidManifest中定义为单独进程）
class GuardianService : Service() {
    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        checkMainProcess()
        return START_STICKY
    }
    
    private fun checkMainProcess() {
        val timer = Timer()
        timer.scheduleAtFixedRate(object : TimerTask() {
            override fun run() {
                if (!isProcessRunning("MainService")) {
                    startService(Intent(this@GuardianService, MainService::class.java))
                }
            }
        }, 0, 5000)
    }
}
```

## 三、权限管理机制

### 3.1 权限分类

```
1. 正常权限（Normal Permissions）
   - 低风险，自动授予
   - INTERNET, ACCESS_NETWORK_STATE, BLUETOOTH等
   
2. 危险权限（Dangerous Permissions）
   - 高风险，需要运行时请求
   - LOCATION, CAMERA, MICROPHONE, READ_CONTACTS等
```

### 3.2 运行时权限请求 - Android 6.0+

**原理**：编译时权限→运行时权限的演进

```kotlin
// AndroidManifest.xml
<uses-permission android:name="android.permission.ACCESS_FINE_LOCATION" />
<uses-permission android:name="android.permission.CAMERA" />

class PermissionManager(private val activity: Activity) {
    private val permissions = arrayOf(
        Manifest.permission.ACCESS_FINE_LOCATION,
        Manifest.permission.CAMERA
    )
    
    private val requestPermissionLauncher = activity.registerForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { permissions ->
        permissions.forEach { (permission, isGranted) ->
            when {
                isGranted -> handlePermissionGranted(permission)
                else -> handlePermissionDenied(permission)
            }
        }
    }
    
    fun requestPermissions() {
        // 检查是否已授予权限
        val needRequest = permissions.filter {
            ContextCompat.checkSelfPermission(activity, it) != PackageManager.PERMISSION_GRANTED
        }
        
        if (needRequest.isNotEmpty()) {
            requestPermissionLauncher.launch(needRequest.toTypedArray())
        }
    }
    
    private fun handlePermissionGranted(permission: String) {
        when (permission) {
            Manifest.permission.ACCESS_FINE_LOCATION -> startLocationUpdates()
            Manifest.permission.CAMERA -> openCamera()
        }
    }
    
    private fun handlePermissionDenied(permission: String) {
        when (permission) {
            Manifest.permission.ACCESS_FINE_LOCATION -> 
                Toast.makeText(activity, "需要位置权限", Toast.LENGTH_SHORT).show()
        }
    }
}
```

#### 原理：内存泄漏的四大罪犯

```kotlin
// 1. 静态引用持有Context
class Singleton {
    private val context: Context = null  // 错误❌
    // 原理：Activity销毁时Context仍被静态引用持有，无法GC
}

// 2. 内部类持有外部类引用
class Activity : AppCompatActivity() {
    inner class LoadTask : AsyncTask<>() {
        override fun doInBackground(vararg params: Void?): Result {
            return null
        }
    }
    // 原理：内部类隐式持有outer this，Activity销毁时LoadTask仍可能执行，导致泄漏
}

// 3. 监听器未注销
class MyActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        // 错误❌
        val listener = ValueEventListener { /* ... */ }
        FirebaseDatabase.getInstance().reference.addListenerForSingleValueEvent(listener)
        // 原理：数据库持有listener引用，Activity销毁时listener仍活跃
    }
}

// 4. Timer/Handler未取消
class MyActivity : AppCompatActivity() {
    private val handler = Handler()
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        // 错误❌
        handler.postDelayed({ /* ... */ }, 10000)
        // 如果Activity在10秒内销毁，Handler仍持有Activity引用
    }
}
```

#### 正确的内存管理：

```kotlin
class MemorySafeActivity : AppCompatActivity() {
    private var timer: Timer? = null
    private val handler = Handler(Looper.getMainLooper())
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        // ✓ 使用弱引用
        val weakActivity = WeakReference(this)
        
        // ✓ 定时任务
        timer = Timer()
        timer?.scheduleAtFixedRate(object : TimerTask() {
            override fun run() {
                weakActivity.get()?.updateUI()
            }
        }, 0, 1000)
        
        // ✓ Handler延迟任务
        handler.postDelayed({
            if (!isDestroyed) {  // 检查Activity是否销毁
                updateData()
            }
        }, 1000)
    }
    
    override fun onDestroy() {
        super.onDestroy()
        // ✓ 及时释放资源
        timer?.cancel()
        handler.removeCallbacksAndMessages(null)
    }
}
```

### 4.2 ANR（Application Not Responding）排查

**原理**：主线程5秒内无响应系统会弹ANR对话框

```kotlin
class StutterDetector(private val context: Context) {
    // 使用Choreographer检测帧率下降（ANR前兆）
    private val choreographer = Choreographer.getInstance()
    private var frameTime = System.nanoTime()
    
    fun startMonitoring() {
        choreographer.postFrameCallback(object : Choreographer.FrameCallback {
            override fun doFrame(frameTimeNanos: Long) {
                val jankTime = frameTimeNanos - frameTime
                
                // 原理：两帧时间间隔正常为16.67ms
                // 如果>50ms说明有帧率下降（ANR前兆）
                if (jankTime > 50_000_000) {  // 50ms
                    reportJank()
                }
                
                frameTime = frameTimeNanos
                choreographer.postFrameCallback(this)
            }
        })
    }
    
    // 使用Systrace或Perfetto分析
    // systrace --app=com.example.app -o trace.html
}
```

**ANR常见原因与解决：**

```kotlin
// ❌ 原因1：主线程执行IO操作
class BadActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        // 错误❌
        val data = File("/sdcard/data.txt").readText()  // 阻塞主线程
    }
}

// ✓ 解决方案1：使用协程
class GoodActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        lifecycleScope.launch(Dispatchers.IO) {
            val data = File("/sdcard/data.txt").readText()
            withContext(Dispatchers.Main) {
                updateUI(data)
            }
        }
    }
}

// ❌ 原因2：主线程执行数据库查询
// ✓ 解决：使用Room + Coroutines
@Dao
interface UserDao {
    @Query("SELECT * FROM users")
    suspend fun getAllUsers(): List<User>  // suspend函数在默认IO线程执行
}

// ❌ 原因3：主线程做复杂计算
// ✓ 解决：使用线程池
class ComplexCalculation {
    private val executor = Executors.newFixedThreadPool(2)
    
    fun calculateAsync(data: List<Int>, callback: (Int) -> Unit) {
        executor.submit {
            val result = data.sum()  // 耗时计算
            Handler(Looper.getMainLooper()).post {
                callback(result)
            }
        }
    }
}
```

### 4.3 启动速度优化

**原理**：冷启动 = Zygote fork + attachBaseContext + onCreate

#### 启动流程分析：

```
Zygote fork进程 (5-20ms)
  ↓
Application.attachBaseContext() (10-50ms) - ContentProvider初始化
  ↓
Application.onCreate() (50-100ms)
  ↓
首个Activity.onCreate() (50-150ms)
  ↓
首个Activity.onResume() - UI可见
  ↓
首个Activity.onStart() 时渲染第一帧
```

#### 冷启动优化方案：

```kotlin
// AAOS特定：初始化CarApi很耗时，需要异步
class CarServiceInitializer(private val context: Context) {
    private var car: Car? = null
    
    // ✓ 在后台线程初始化Car
    fun initCarAsync(callback: (Car?) -> Unit) {
        Thread {
            try {
                car = Car.createCar(context)
                callback(car)
            } catch (e: Exception) {
                callback(null)
            }
        }.start()
    }
    
    // ✓ 复用Car实例，避免重复创建
    fun getCarInstance(): Car? = car
}
```
