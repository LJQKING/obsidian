
## 一、HTTP 请求优化

### 1. **OkHttp 连接复用（最关键）**

```kotlin
// ✅ 全局单例 OkHttpClient
object HttpClientFactory {
    private val okHttpClient: OkHttpClient by lazy {
        OkHttpClient.Builder()
            .connectTimeout(15, TimeUnit.SECONDS)
            .readTimeout(20, TimeUnit.SECONDS)
            .writeTimeout(20, TimeUnit.SECONDS)
            // 核心：连接池配置
            .connectionPool(ConnectionPool(
                maxIdleConnections = 8,  // 空闲连接数
                keepAliveDuration = 5,
                TimeUnit.MINUTES         // HTTP/1.1 keep-alive
            ))
            // 快速失败
            .retryOnConnectionFailure(true)
            .addInterceptor(LoggingInterceptor())
            .addNetworkInterceptor(ChunkedEncodingInterceptor())
            .build()
    }
}

// ✅ Retrofit 配置
val retrofit = Retrofit.Builder()
    .client(HttpClientFactory.okHttpClient)
    .baseUrl("https://api.example.com/")
    .addConverterFactory(GsonConverterFactory.create())
    .addCallAdapterFactory(RxJava3CallAdapterFactory.create())
    .build()
```

**为什么要连接复用？**

- 避免 TCP 三次握手开销（每个新连接 100ms+）
- HTTP/1.1 Keep-Alive 在同一连接上复用
- 减少 SSL/TLS 握手（HTTPS）

### 2. **缓存策略（两级缓存）**

```kotlin
// ✅ 内存缓存 + 磁盘缓存
fun createOkHttpClientWithCache(): OkHttpClient {
    val cacheDir = File(context.cacheDir, "http_cache")
    val cache = Cache(cacheDir, 50 * 1024 * 1024) // 50MB
    
    return OkHttpClient.Builder()
        .cache(cache)
        .addNetworkInterceptor(CacheInterceptor())
        .build()
}

// ✅ 自定义缓存拦截器
class CacheInterceptor : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val request = chain.request()
        val response = chain.proceed(request)
        
        return if (request.method == "GET") {
            response.newBuilder()
                .header("Cache-Control", "public, max-age=300") // 5分钟
                .build()
        } else {
            response // POST 不缓存
        }
    }
}
```

### 2. **Retrofit @POST 问题示例**

```kotlin
// ❌ 问题：没有处理大 Body
@POST("/api/upload")
fun uploadData(@Body data: String): Single<UploadResponse>

// ✅ 解决：分块上传
@POST("/api/upload/chunk")
@Multipart
fun uploadChunk(
    @Part("file") file: RequestBody,
    @Part("chunkId") chunkId: Int,
    @Part("totalChunks") totalChunks: Int
): Single<UploadResponse>

// 实现
suspend fun uploadLargeFile(file: File, chunkSize: Int = 512 * 1024) {
    val totalChunks = (file.length() / chunkSize).toInt() + 1
    
    file.inputStream().use { input ->
        var chunkId = 0
        val buffer = ByteArray(chunkSize)
        var bytesRead: Int
        
        while (input.read(buffer).also { bytesRead = it } > 0) {
            val chunk = buffer.sliceArray(0 until bytesRead)
            val requestBody = chunk.toRequestBody("application/octet-stream".toMediaType())
            
            apiService.uploadChunk(requestBody, chunkId, totalChunks)
                .await()
            
            chunkId++
        }
    }
}
```

### 3. **超时问题处理**

```kotlin
// ✅ 分级超时策略
fun createOkHttpClient(): OkHttpClient {
    return OkHttpClient.Builder()
        .callTimeout(60, TimeUnit.SECONDS)      // 总超时
        .connectTimeout(10, TimeUnit.SECONDS)   // 连接超时
        .readTimeout(20, TimeUnit.SECONDS)      // 读超时
        .writeTimeout(20, TimeUnit.SECONDS)     // 写超时
        .addInterceptor { chain ->
            val originalRequest = chain.request()
            
            // 为长时间操作延长超时
            val newRequest = if (originalRequest.header("X-Long-Operation") != null) {
                chain.withConnectTimeout(30, TimeUnit.SECONDS)
                    .withReadTimeout(120, TimeUnit.SECONDS)
                    .withWriteTimeout(120, TimeUnit.SECONDS)
                    .proceed(originalRequest)
            } else {
                chain.proceed(originalRequest)
            }
            
            newRequest
        }
        .build()
}
```

### 4. **请求限流（Rate Limiting）**

```kotlin
// ✅ 实现请求队列和限流
class RateLimitInterceptor(private val maxRequests: Int = 5) : Interceptor {
    private val requestSemaphore = Semaphore(maxRequests)
    
    override fun intercept(chain: Interceptor.Chain): Response {
        requestSemaphore.acquire()
        try {
            return chain.proceed(chain.request())
        } finally {
            requestSemaphore.release()
        }
    }
}

// ✅ 使用 OkHttp 的 Dispatcher 限流
okHttpClient.dispatcher.apply {
    maxRequests = 5          // 全局最大请求数
    maxRequestsPerHost = 3   // 单个主机最大请求数
}
```

### 5. **Gzip 压缩**

```kotlin
// ✅ 自动启用 Gzip（OkHttp 内置）
// OkHttp 会自动添加 Accept-Encoding: gzip
// 但对于 POST body 需要手动处理

@POST("/api/data")
fun uploadCompressed(@Body data: String): Single<Response>

// 拦截器自动压缩
class GzipInterceptor : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val originalRequest = chain.request()
        
        val compressedBody = originalRequest.body?.let {
            GzipRequestBody(it)
        } ?: return chain.proceed(originalRequest)
        
        val compressedRequest = originalRequest.newBuilder()
            .header("Content-Encoding", "gzip")
            .method(originalRequest.method, compressedBody)
            .build()
        
        return chain.proceed(compressedRequest)
    }
}
```
