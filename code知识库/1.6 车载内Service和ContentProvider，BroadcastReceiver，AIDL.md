## 一、Service：长连接 + 后台计算

### 场景：HVAC（空调控制）服务，HMI App 绑定后实时读写温度

```kotlin
// ── ICarHvacService.aidl ──────────────────────────────────
// packages/apps/Car/HVAC/src/com/example/hvac/ICarHvacService.aidl
package com.example.hvac;

import com.example.hvac.IHvacCallback;

interface ICarHvacService {
    // 设置驾驶侧温度（支持 ZONE_ROW_1_LEFT / RIGHT）
    void setTemperature(int areaId, float tempCelsius);
    float getTemperature(int areaId);
    void registerCallback(IHvacCallback callback);
    void unregisterCallback(IHvacCallback callback);
```

```kotlin
// ── HvacService.kt（运行在独立进程 android:process=":hvac"）──
class HvacService : Service() {

    // RemoteCallbackList 线程安全地管理跨进程 callback
    private val callbacks = RemoteCallbackList<IHvacCallback>()

    // 与 VHAL 通信：CarPropertyManager 是 AAOS 标准接口
    private lateinit var carPropertyManager: CarPropertyManager
    private lateinit var car: Car

    private val binder = object : ICarHvacService.Stub() {

        override fun setTemperature(areaId: Int, tempCelsius: Float) {
            // Binder 线程池中执行，需切回主线程操作 CarProperty
            carPropertyManager.setFloatProperty(
                VehiclePropertyIds.HVAC_TEMPERATURE_SET,
                areaId,
                tempCelsius
            )
            // 通知所有已注册的 HMI 组件
            notifyTemperatureChanged(areaId, tempCelsius)
        }

        override fun getTemperature(areaId: Int): Float =
            carPropertyManager.getFloatProperty(
                VehiclePropertyIds.HVAC_TEMPERATURE_CURRENT, areaId
            )

        override fun registerCallback(callback: IHvacCallback) {
            callbacks.register(callback)
        }

        override fun unregisterCallback(callback: IHvacCallback) {
            callbacks.unregister(callback)
        }
    }

    override fun onCreate() {
        super.onCreate()
        // Car.createCar 是 AAOS 接入 CarService 的标准姿势
        car = Car.createCar(this)
        carPropertyManager = car.getCarManager(Car.PROPERTY_SERVICE) as CarPropertyManager

        // 监听 VHAL 上报的温度变化（CAN 总线 → VHAL → CarPropertyManager）
        carPropertyManager.registerCallback(
            object : CarPropertyManager.CarPropertyEventCallback {
                override fun onChangeEvent(value: CarPropertyValue<*>) {
                    if (value.propertyId == VehiclePropertyIds.HVAC_TEMPERATURE_CURRENT) {
                        notifyTemperatureChanged(value.areaId, value.value as Float)
                    }
                }
                override fun onErrorEvent(propId: Int, zone: Int) {}
            },
            VehiclePropertyIds.HVAC_TEMPERATURE_CURRENT,
            CarPropertyManager.SENSOR_RATE_ONCHANGE
        )
    }

    // 遍历广播给所有客户端（跨进程安全）
    private fun notifyTemperatureChanged(areaId: Int, temp: Float) {
        val n = callbacks.beginBroadcast()
        repeat(n) { i ->
            try {
                callbacks.getBroadcastItem(i).onTemperatureChanged(areaId, temp)
            } catch (e: RemoteException) { /* 客户端已死，RemoteCallbackList 自动清理 */ }
        }
        callbacks.finishBroadcast()
    }

    override fun onBind(intent: Intent): IBinder = binder

    override fun onDestroy() {
        car.disconnect()
        super.onDestroy()
    }
}
```

```kotlin
// ── HMI 侧绑定（空调控制 Fragment）──
```

**原理要点：**

|问题|机制|
|---|---|
|为何用 `RemoteCallbackList`|自动处理客户端进程死亡时的 IBinder 清理，避免 `DeadObjectException`|
|`android:process=":hvac"`|独立进程保证 HVAC 逻辑与 UI 隔离，系统 kill UI 进程不影响温控|
|Binder 线程安全|`Stub` 方法在 Binder 线程池执行，CarProperty 操作已有内部锁，回调 UI 必须 `runOnUiThread`|
|VHAL 链路|`setTemperature` → `CarPropertyManager` → `CarService` → **VHAL HAL** → CAN 驱动 → 整车总线|

## 二、ContentProvider：跨进程结构化数据共享

### 场景：车辆状态数据中心，多个 App（导航、仪表、HUD）统一读取行驶数据

```kotlin
// ── VehicleDataProvider.kt ──
@Singleton
class VehicleDataProvider : ContentProvider() {

    companion object {
        const val AUTHORITY = "com.example.car.vehicle_data"
        // content://com.example.car.vehicle_data/speed
        // content://com.example.car.vehicle_data/fuel
        // content://com.example.car.vehicle_data/gear
        val SPEED_URI: Uri = Uri.parse("content://$AUTHORITY/speed")
        val FUEL_URI:  Uri = Uri.parse("content://$AUTHORITY/fuel")
        val GEAR_URI:  Uri = Uri.parse("content://$AUTHORITY/gear")

        private val uriMatcher = UriMatcher(UriMatcher.NO_MATCH).apply {
            addURI(AUTHORITY, "speed", 1)
            addURI(AUTHORITY, "fuel",  2)
            addURI(AUTHORITY, "gear",  3)
            addURI(AUTHORITY, "speed/#", 4) // 按 areaId 精确查询
        }
    }

    // 内存缓存（也可换 Room DB 做持久化）
    private val vehicleCache = ConcurrentHashMap<String, Any>()
    private lateinit var carPropertyManager: CarPropertyManager

    override fun onCreate(): Boolean {
        val car = Car.createCar(context)
        carPropertyManager = car.getCarManager(Car.PROPERTY_SERVICE) as CarPropertyManager

        // 订阅 VHAL，更新本地缓存并通知订阅者
        carPropertyManager.registerCallback(vhalCallback,
            VehiclePropertyIds.PERF_VEHICLE_SPEED, CarPropertyManager.SENSOR_RATE_FAST)
        carPropertyManager.registerCallback(vhalCallback,
            VehiclePropertyIds.FUEL_LEVEL, CarPropertyManager.SENSOR_RATE_ONCHANGE)
        carPropertyManager.registerCallback(vhalCallback,
            VehiclePropertyIds.GEAR_SELECTION, CarPropertyManager.SENSOR_RATE_ONCHANGE)
        return true
    }

    private val vhalCallback = object : CarPropertyManager.CarPropertyEventCallback {
        override fun onChangeEvent(value: CarPropertyValue<*>) {
            when (value.propertyId) {
                VehiclePropertyIds.PERF_VEHICLE_SPEED -> {
                    vehicleCache["speed"] = value.value as Float
                    context?.contentResolver?.notifyChange(SPEED_URI, null)
                }
                VehiclePropertyIds.FUEL_LEVEL -> {
                    vehicleCache["fuel"] = value.value as Float
                    context?.contentResolver?.notifyChange(FUEL_URI, null)
                }
                VehiclePropertyIds.GEAR_SELECTION -> {
                    vehicleCache["gear"] = value.value as Int
                    context?.contentResolver?.notifyChange(GEAR_URI, null)
                }
            }
        }
        override fun onErrorEvent(propId: Int, zone: Int) {}
    }

    override fun query(
        uri: Uri, projection: Array<String>?, selection: String?,
        selectionArgs: Array<String>?, sortOrder: String?
    ): Cursor {
        val cursor = MatrixCursor(arrayOf("key", "value", "timestamp"))
        when (uriMatcher.match(uri)) {
            1 -> cursor.addRow(arrayOf("speed",
                    vehicleCache["speed"] ?: 0f, System.currentTimeMillis()))
            2 -> cursor.addRow(arrayOf("fuel",
                    vehicleCache["fuel"]  ?: 0f, System.currentTimeMillis()))
            3 -> cursor.addRow(arrayOf("gear",
                    vehicleCache["gear"]  ?: 0,  System.currentTimeMillis()))
        }
        // 关键：让 Cursor 订阅 Uri 变化，客户端 CursorLoader 自动刷新
        cursor.setNotificationUri(context!!.contentResolver, uri)
        return cursor
    }

    // ContentProvider 不支持写整车属性（只读），insert/update/delete 返回 0
    override fun getType(uri: Uri): String = "vnd.android.cursor.item/vehicle_data"
    override fun insert(uri: Uri, values: ContentValues?): Uri? = null
    override fun delete(uri: Uri, selection: String?, selectionArgs: Array<String>?) = 0
    override fun update(uri: Uri, values: ContentValues?,
                        selection: String?, selectionArgs: Array<String>?) = 0
}
```

```kotlin
// ── 仪表盘 App 订阅速度（使用 ContentObserver）──
class InstrumentClusterFragment : Fragment() {

    private val speedObserver = object : ContentObserver(Handler(Looper.getMainLooper())) {
        override fun onChange(selfChange: Boolean) {
            // notifyChange 触发，在主线程直接查询并更新 UI
            val speed = querySpeed()
            binding.speedometer.setValue(speed)
        }
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        // 注册 ContentObserver，第三个参数 true = 监听子 Uri
        requireContext().contentResolver.registerContentObserver(
            VehicleDataProvider.SPEED_URI, true, speedObserver
        )
        // 首次加载当前值
        binding.speedometer.setValue(querySpeed())
    }

    private fun querySpeed(): Float {
        val cursor = requireContext().contentResolver.query(
            VehicleDataProvider.SPEED_URI,
            null, null, null, null
        ) ?: return 0f
        return cursor.use {
            if (it.moveToFirst()) it.getFloat(it.getColumnIndexOrThrow("value")) else 0f
        }
    }

    override fun onDestroyView() {
        requireContext().contentResolver.unregisterContentObserver(speedObserver)
        super.onDestroyView()
    }
}
```

**原理要点：**

|问题|机制|
|---|---|
|为何用 CP 而不直接 AIDL|CP 有标准 `CursorLoader` + `ContentObserver` 机制，**多个 App 零耦合订阅**同一份数据|
|`notifyChange` 链路|VHAL 回调 → 更新缓存 → `notifyChange(uri)` → 所有注册 `ContentObserver` 的进程被唤醒|
|`setNotificationUri`|Cursor 持有 Uri 引用，`CursorLoader` 在 `notifyChange` 时自动重新 query|
|权限声明|Manifest 中 `<provider android:readPermission="com.example.car.READ_VEHICLE"/>` 防止第三方读取|

## 三、BroadcastReceiver：系统事件响应

### 场景1：整车上电启动序列（CarBoot → 各服务就绪）

```kotlin
// ── CarBootReceiver.kt（静态注册，支持进程未启动时被唤醒）──
class CarBootReceiver : BroadcastReceiver() {

    override fun onReceive(context: Context, intent: Intent) {
        when (intent.action) {
            // AAOS 专有：整车 ACC ON（钥匙拨到 ON 挡）
            CarPowerStateListener.ACTION_CAR_POWER_STATE_CHANGE -> {
                val state = intent.getIntExtra(
                    CarPowerStateListener.EXTRA_CAR_POWER_STATE, -1)
                when (state) {
                    CarPowerStateListener.ON -> {
                        // 启动 HVAC / Media / Navigation 服务
                        startCarServices(context)
                    }
                    CarPowerStateListener.SHUTDOWN_PREPARE -> {
                        // 整车下电前保存用户偏好
                        saveUserPreferences(context)
                    }
                }
            }
            // 标准 Android：系统重启后自动拉起座舱服务
            Intent.ACTION_BOOT_COMPLETED -> {
                startCarServices(context)
            }
        }
    }

    private fun startCarServices(context: Context) {
        // 使用 JobScheduler 而非直接 startService（Android 8+ 后台限制）
        val scheduler = context.getSystemService(JobScheduler::class.java)
        val jobInfo = JobInfo.Builder(1001,
            ComponentName(context, CarServiceInitJob::class.java))
            .setOverrideDeadline(0) // 立即执行
            .build()
        scheduler.schedule(jobInfo)
    }
}
```

```kotlin
    context.sendOrderedBroadcast(

    )
```

```kotlin
// ── PhoneLinkService.kt（实现 AIDL Stub，平台适配层）──
class PhoneLinkService : Service() {

    private val callbacks = RemoteCallbackList<IPhoneLinkCallback>()

    // AIDL 核心实现
    private val binder = object : IPhoneLinkService.Stub() {

        // oneway 方法：调用端不等待，在 Binder 线程异步执行
        override fun startProjection(session: PhoneLinkSession) {
            // 根据协议类型路由到不同适配器（AdaptApi 设计模式）
            val adapter: ProtocolAdapter = when (session.protocol) {
                "HICAR"    -> HiCarAdapter(session)
                "CARPLAY"  -> CarPlayAdapter(session)
                "CARLINK"  -> CarLinkAdapter(session)
                else       -> throw IllegalArgumentException("Unknown protocol")
            }
            adapter.startSession()
            notifyConnectionState(STATE_CONNECTED)
        }

        override fun stopProjection(sessionId: String) {
            AdapterRegistry.getAdapter(sessionId)?.stopSession()
            notifyConnectionState(STATE_DISCONNECTED)
        }

        // 同步调用：直接返回，Binder 驱动负责序列化
        override fun getConnectionState(): Int =
            AdapterRegistry.currentState

        override fun registerCallback(callback: IPhoneLinkCallback) {
            callbacks.register(callback)
        }

        override fun unregisterCallback(callback: IPhoneLinkCallback) {
            callbacks.unregister(callback)
        }
    }

    private fun notifyConnectionState(state: Int) {
        val n = callbacks.beginBroadcast()
        repeat(n) { i ->
            try { callbacks.getBroadcastItem(i).onConnectionStateChanged(state) }
            catch (_: RemoteException) {}
        }
        callbacks.finishBroadcast()
    }

    override fun onBind(intent: Intent): IBinder {
        // 权限校验：只有持有特定权限的调用方才允许绑定
        if (checkCallingPermission("com.example.car.PHONE_LINK") !=
            PackageManager.PERMISSION_GRANTED) {
            throw SecurityException("Missing PHONE_LINK permission")
        }
        return binder
    }
}
```

## 面试高频问题速查

| 问题                                        | 核心答案                                                                                                  |
| ----------------------------------------- | ----------------------------------------------------------------------------------------------------- |
| Service `started` vs `bound` 区别           | `started`：`startService`，独立生命周期；`bound`：`bindService`，与客户端共存亡                                         |
| AIDL `oneway` 关键字                         | 异步非阻塞，调用方无需等待，**不能有返回值**；适合高频单向数据投递                                                                   |
| CP `notifyChange` 原理                      | `ContentResolver` 通过 Binder 通知所有注册 `ContentObserver` 的进程，底层是 `ActivityManagerService` 维护的 observer 列表 |
| 有序广播 vs 普通广播                              | 有序：串行按优先级，可 `abortBroadcast()`，可附加结果；普通：并行，无优先级                                                       |
| Binder 最大数据量                              | 单次 Binder 事务约 **1MB**，大数据（图像帧）用共享内存（`MemoryFile` / `Ashmem`）传递，AIDL 中声明为 `FileDescriptor`             |
| 为什么 AAOS 用 `CarPropertyManager` 而不直接 AIDL | `CarPropertyManager` 封装了 VHAL 的 HAL 接口，内置区域（Zone）、采样率（SensorRate）、权限校验，是 AAOS 的标准抽象层                  |
