
## 一、缓存策略实现

### 1. 内存缓存 + 数据库缓存的二级缓存方案

```kotlin
/**
 * 两级缓存策略
 */
class CacheStrategy {
    // 内存缓存
    private val memoryCache = mutableMapOf<String, CacheData<*>>()
    
    // 数据库缓存 (Room)
    @Inject
    lateinit var cacheDao: CacheDao
    
    data class CacheData<T>(
        val data: T,
        val timestamp: Long,
        val ttl: Long // 生存时间(毫秒)
    ) {
        fun isExpired(): Boolean = System.currentTimeMillis() - timestamp > ttl
    }
    
    /**
     * 从两级缓存获取数据
     */
    suspend fun <T> getFromCache(key: String, clazz: Class<T>): T? {
        // 1. 检查内存缓存
        memoryCache[key]?.let { cache ->
            @Suppress("UNCHECKED_CAST")
            val cacheData = cache as CacheData<T>
            if (!cacheData.isExpired()) {
                return cacheData.data
            } else {
                memoryCache.remove(key)
            }
        }
        
        // 2. 检查数据库缓存
        val dbCache = cacheDao.getCache(key)
        if (dbCache != null && !dbCache.isExpired()) {
            val data = Gson().fromJson(dbCache.data, clazz)
            // 回写内存缓存
            putToMemoryCache(key, data, dbCache.ttl)
            return data
        }
        
        return null
    }
    
    /**
     * 存入两级缓存
     */
    suspend fun <T> putToCache(
        key: String,
        data: T,
        ttlMinutes: Long = 30
    ) {
        val ttl = ttlMinutes * 60 * 1000
        putToMemoryCache(key, data, ttl)
        putToDbCache(key, data, ttl)
    }
    
    private fun <T> putToMemoryCache(key: String, data: T, ttl: Long) {
        memoryCache[key] = CacheData(data, System.currentTimeMillis(), ttl)
    }
    
    private suspend fun <T> putToDbCache(key: String, data: T, ttl: Long) {
        val jsonData = Gson().toJson(data)
        cacheDao.insertCache(
            CacheEntity(
                key = key,
                data = jsonData,
                timestamp = System.currentTimeMillis(),
                ttl = ttl
            )
        )
    }
    
    /**
     * 清空缓存
     */
    suspend fun clearCache() {
        memoryCache.clear()
        cacheDao.deleteAllCache()
    }
}

// Room数据库定义
@Entity(tableName = "cache_table")
data class CacheEntity(
    @PrimaryKey val key: String,
    val data: String,
    val timestamp: Long,
    val ttl: Long
) {
    fun isExpired(): Boolean = System.currentTimeMillis() - timestamp > ttl
}

@Dao
interface CacheDao {
    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertCache(cache: CacheEntity)
    
    @Query("SELECT * FROM cache_table WHERE key = :key")
    suspend fun getCache(key: String): CacheEntity?
    
    @Query("DELETE FROM cache_table")
    suspend fun deleteAllCache()
}
```

### 2. 网络优先vs缓存优先的混合策略

```kotlin
/**
 * 混合缓存策略的Repository
 * - 优先返回缓存，后台更新网络数据
 * - 网络失败自动降级到缓存
 */
class UserRepositoryWithCache @Inject constructor(
    private val apiService: UserApiService,
    private val cacheStrategy: CacheStrategy
) {
    
    /**
     * 策略1: 缓存优先 + 后台更新（推荐用于列表页）
     */
    fun getUserListCacheFirst(page: Int, pageSize: Int): Observable<UserListResponse> {
        val cacheKey = "user_list_${page}_${pageSize}"
        
        return Observable.create { emitter ->
            // 1. 先发送缓存数据
            runBlocking {
                cacheStrategy.getFromCache(
                    cacheKey,
                    UserListResponse::class.java
                )?.let {
                    emitter.onNext(it)
                }
            }
            
            // 2. 后台更新网络数据
            apiService.getUserList(page, pageSize)
                .compose(RxJavaUtils.handleApiResponse())
                .subscribe(
                    { data ->
                        // 缓存新数据
                        runBlocking {
                            cacheStrategy.putToCache(cacheKey, data, ttlMinutes = 30)
                        }
                        emitter.onNext(data)
                        emitter.onComplete()
                    },
                    { error ->
                        // 网络失败，但缓存已返回则成功，否则失败
                        if (!emitter.isDisposed) {
                            emitter.onError(error)
                        }
                    }
                )
        }
        .compose(RxJavaUtils.ioToMain())
    }
    
    /**
     * 策略2: 网络优先 + 降级到缓存（用于详情页）
     */
    fun getUserDetailNetworkFirst(userId: Long): Observable<User> {
        val cacheKey = "user_detail_$userId"
        
        return apiService.getUserDetail(userId)
            .compose(RxJavaUtils.handleApiResponse())
            .doOnNext { user ->
                runBlocking {
                    cacheStrategy.putToCache(cacheKey, user, ttlMinutes = 60)
                }
            }
            .onErrorResumeNext { error ->
                // 网络失败，降级到缓存
                runBlocking {
                    cacheStrategy.getFromCache(cacheKey, User::class.java)?.let {
                        Observable.just(it)
                    } ?: Observable.error(error)
                }
            }
            .compose(RxJavaUtils.ioToMain())
    }
    
    /**
     * 策略3: 仅网络（用于实时数据）
     */
    fun getUserDetailNetworkOnly(userId: Long): Observable<User> {
        return apiService.getUserDetail(userId)
            .compose(RxJavaUtils.handleApiResponse())
            .compose(RxJavaUtils.ioToMain())
    }
}
```

```kotlin
/**
 * 网络状态管理
 */
class NetworkStateManager @Inject constructor(
    context: Context
) {
    
    private val connectivityManager = 
        context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
    
    private val _networkState = MutableSharedFlow<NetworkState>(replay = 1)
    val networkState = _networkState.asSharedFlow()
    
    enum class NetworkState {
        CONNECTED,      // 已连接
        DISCONNECTED,   // 已断开
        CONNECTING      // 连接中
    }
    
    init {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
            connectivityManager.registerDefaultNetworkCallback(
                object : ConnectivityManager.NetworkCallback() {
                    override fun onAvailable(network: Network) {
                        super.onAvailable(network)
                        GlobalScope.launch {
                            _networkState.emit(NetworkState.CONNECTED)
                        }
                    }
                    
                    override fun onLost(network: Network) {
                        super.onLost(network)
                        GlobalScope.launch {
                            _networkState.emit(NetworkState.DISCONNECTED)
                        }
                    }
                }
            )
        }
    }
    
    /**
     * 检查当前网络连接状态
     */
    fun isConnected(): Boolean {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            val network = connectivityManager.activeNetwork
            val capabilities = connectivityManager.getNetworkCapabilities(network)
            capabilities?.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET) == true
        } else {
            val info = connectivityManager.activeNetworkInfo
            info?.isConnected == true
        }
    }
    
    /**
     * 获取网络类型
     */
    fun getNetworkType(): String {
        return when {
            Build.VERSION.SDK_INT >= Build.VERSION_CODES.M -> {
                val network = connectivityManager.activeNetwork
                val capabilities = connectivityManager.getNetworkCapabilities(network)
                when {
                    capabilities?.hasTransport(NetworkCapabilities.TRANSPORT_WIFI) == true -> "WiFi"
                    capabilities?.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR) == true -> "Cellular"
                    capabilities?.hasTransport(NetworkCapabilities.TRANSPORT_BLUETOOTH) == true -> "Bluetooth"
                    else -> "Unknown"
                }
            }
            else -> {
                val info = connectivityManager.activeNetworkInfo
                when (info?.type) {
                    ConnectivityManager.TYPE_WIFI -> "WiFi"
                    ConnectivityManager.TYPE_MOBILE -> "Cellular"
                    else -> "Unknown"
                }
            }
        }
    }
}
```

### 2. 离线模式下的请求处理

```kotlin
/**
 * 离线模式拦截器
 * - 断网时，返回缓存数据
 * - 支持离线操作队列
 */
class OfflineInterceptor @Inject constructor(
    private val context: Context,
    private val networkStateManager: NetworkStateManager,
    private val cacheStrategy: CacheStrategy
) : Interceptor {
    
    override fun intercept(chain: Interceptor.Chain): Response {
        val request = chain.request()
        
        // 如果有网络，正常请求
        if (networkStateManager.isConnected()) {
            return try {
                chain.proceed(request)
            } catch (e: Exception) {
                // 网络错误，尝试返回缓存
                handleOfflineMode(request)
            }
        }
        
        // 断网状态，尝试返回缓存
        return handleOfflineMode(request)
    }
    
    private fun handleOfflineMode(request: Request): Response {
        val url = request.url.toString()
        val cacheKey = generateCacheKey(request)
        
        // 尝试从缓存获取数据
        val cachedData = runBlocking {
            // 这里简化处理，实际应该根据类型解析
            cacheStrategy.getFromCache(cacheKey, String::class.java)
        }
        
        return if (cachedData != null) {
            // 返回缓存数据
            Response.Builder()
                .code(200)
                .message("Offline - from cache")
                .protocol(Protocol.HTTP_1_1)
                .request(request)
                .body(ResponseBody.create(
                    MediaType.parse("application/json"),
                    cachedData
                ))
                .header("X-From-Cache", "true")
                .build()
        } else {
            // 没有缓存，返回错误
            Response.Builder()
                .code(503)
                .message("Service Unavailable - Offline")
                .protocol(Protocol.HTTP_1_1)
                .request(request)
                .body(ResponseBody.create(
                    MediaType.parse("application/json"),
                    """{"code":503,"message":"离线状态，无缓存数据"}"""
                ))
                .build()
        }
    }
    
    private fun generateCacheKey(request: Request): String {
        return request.url.toString().hashCode().toString()
    }
}

// 在OkHttp配置中添加
class OkHttpClientFactory {
    fun create(

        cacheStrategy: CacheStrategy
    ): OkHttpClient {
        return OkHttpClient.Builder()
            // ... 其他配置
            .addInterceptor(OfflineInterceptor(context, networkStateManager, cacheStrategy))
            .build()
    }
}
```

```kotlin
/**
 * 离线操作队列
 * 用于缓存用户在离线时的操作，网络恢复后自动同步
 */
@Entity(tableName = "offline_queue")
data class OfflineOperation(
    @PrimaryKey(autoGenerate = true) val id: Int = 0,
    val method: String,  // POST/PUT/DELETE
    val url: String,
    val body: String,
    val timestamp: Long,
    val retryCount: Int = 0,
    val status: String = "pending"  // pending/success/failed
)

@Dao
interface OfflineQueueDao {
    @Insert
    suspend fun insertOperation(operation: OfflineOperation): Long
    
    @Query("SELECT * FROM offline_queue WHERE status = 'pending' ORDER BY timestamp ASC")
    suspend fun getPendingOperations(): List<OfflineOperation>
    
    @Update
    suspend fun updateOperation(operation: OfflineOperation)
    
    @Delete
    suspend fun deleteOperation(operation: OfflineOperation)
}

/**
 * 离线队列管理器
 */
class OfflineQueueManager @Inject constructor(
    @Inject private val queueDao: OfflineQueueDao,
    private val networkStateManager: NetworkStateManager,
    private val apiService: UserApiService
) {
    
    private val scope = CoroutineScope(Dispatchers.IO + Job())
    
    init {
        // 监听网络状态，恢复时同步队列
        scope.launch {
            networkStateManager.networkState.collect { state ->
                if (state == NetworkStateManager.NetworkState.CONNECTED) {
                    syncOfflineOperations()
                }
            }
        }
    }
    
    /**
     * 添加操作到离线队列
     */
    suspend fun enqueueOperation(
        method: String,
        url: String,
        body: Any? = null
    ): Long {
        val bodyJson = if (body != null) Gson().toJson(body) else ""
        return queueDao.insertOperation(
            OfflineOperation(
                method = method,
                url = url,
                body = bodyJson,
                timestamp = System.currentTimeMillis()
            )
        )
    }
    
    /**
     * 同步所有离线操作
     */
    private suspend fun syncOfflineOperations() {
        val operations = queueDao.getPendingOperations()
        
        for (operation in operations) {
            try {
                // 这里简化处理，实际应该根据method类型调用相应接口
                when (operation.method) {
                    "POST" -> syncPostOperation(operation)
                    "PUT" -> syncPutOperation(operation)
                    "DELETE" -> syncDeleteOperation(operation)
                }
                
                // 更新状态
                queueDao.updateOperation(
                    operation.copy(status = "success")
                )
            } catch (e: Exception) {
                Log.e("OfflineQueue", "Sync failed", e)
                // 重试次数限制
                if (operation.retryCount < 3) {
                    queueDao.updateOperation(
                        operation.copy(retryCount = operation.retryCount + 1)
                    )
                } else {
                    queueDao.updateOperation(
                        operation.copy(status = "failed")
                    )
                }
            }
        }
    }
    
    private suspend fun syncPostOperation(operation: OfflineOperation) {
        val user = Gson().fromJson(operation.body, User::class.java)
        apiService.createUser(user).firstOrError().await()
    }
    
    private suspend fun syncPutOperation(operation: OfflineOperation) {
        val userId = extractUserIdFromUrl(operation.url)
        val user = Gson().fromJson(operation.body, User::class.java)
        apiService.updateUser(userId, user).firstOrError().await()
    }
    
    private suspend fun syncDeleteOperation(operation: OfflineOperation) {
        val userId = extractUserIdFromUrl(operation.url)
        apiService.deleteUser(userId).firstOrError().await()
    }
    
    private fun extractUserIdFromUrl(url: String): Long {
        return url.split("/").last().toLong()
    }
    
    fun destroy() {
        scope.cancel()
    }
}
```

### 1. 请求限流（Rate Limiting）

```kotlin
/**
 * 漏桶算法实现的请求限流
 */
class RateLimiter(
    private val maxRequests: Int = 10,
    private val windowMs: Long = 1000
) {
    private val requestTimestamps = ArrayDeque<Long>()
    private val lock = ReentrantLock()
    
    /**
     * 尝试获取请求许可
     * true: 可以请求，false: 受限中
     */
    fun tryAcquire(): Boolean {
        lock.withLock {
            val now = System.currentTimeMillis()
            val cutoffTime = now - windowMs
            
            // 移除过期的请求时间戳
            while (requestTimestamps.isNotEmpty() && 
                   requestTimestamps.first < cutoffTime) {
                requestTimestamps.removeFirst()
            }
            
            return if (requestTimestamps.size < maxRequests) {
                requestTimestamps.addLast(now)
                true
            } else {
                false
            }
        }
    }
    
    /**
     * 阻塞式获取许可
     */
    suspend fun acquire() {
        while (!tryAcquire()) {
            delay(10)
        }
    }
}

/**
 * 限流拦截器
 */
class RateLimitingInterceptor(
    private val rateLimiter: RateLimiter = RateLimiter(
        maxRequests = 20,
        windowMs = 1000
    )
) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val allowed = rateLimiter.tryAcquire()
        
        if (!allowed) {
            // 返回429 Too Many Requests
            return Response.Builder()
                .code(429)
                .message("Too Many Requests")
                .protocol(Protocol.HTTP_1_1)
                .request(chain.request())
                .body(ResponseBody.create(
                    MediaType.parse("application/json"),
                    """{"code":429,"message":"请求过于频繁"}"""
                ))
                .header("Retry-After", "1")
                .build()
        }
        
        return chain.proceed(chain.request())
    }
}
```

### 2. 并发请求管理（并发度控制）

```kotlin
/**
 * 并发度控制 - 一次最多同时执行N个请求
 */
class ConcurrencyLimiter(private val maxConcurrent: Int = 5) {
    private val activeCount = AtomicInteger(0)
    private val semaphore = Semaphore(maxConcurrent)
    
    /**
     * 包装Observable，加入并发控制
     */
    fun <T> limit(source: Observable<T>): Observable<T> {
        return Observable.create { emitter ->
            source.subscribe(
                { value ->
                    emitter.onNext(value)
                },
                { error ->
                    emitter.onError(error)
                },
                {
                    emitter.onComplete()
                }
            )
        }
        .doOnSubscribe {
            semaphore.acquire()
            activeCount.incrementAndGet()
            Log.d("Concurrency", "Active requests: ${activeCount.get()}")
        }
        .doFinally {
            activeCount.decrementAndGet()
            semaphore.release()
            Log.d("Concurrency", "Active requests: ${activeCount.get()}")
        }
    }
    
    /**
     * 批量请求，控制并发度
     */
    fun <T> limitConcurrentRequests(
        requests: List<Observable<T>>,
        maxConcurrent: Int = this.maxConcurrent
    ): Observable<T> {
        return Observable.fromIterable(requests)
            .flatMap(
                { it },
                maxConcurrent  // 关键参数：并发度
            )
    }
}

// 使用示例
class UserRepositoryWithConcurrencyControl @Inject constructor(
```

```kotlin
class PerformanceMonitoringInterceptor : Interceptor {
    
    data class RequestMetrics(
        val url: String,
        val method: String,
        val statusCode: Int,
        val requestTime: Long,
        val responseTime: Long,

    override fun intercept(chain: Interceptor.Chain): Response {

        val startTime = System.currentTimeMillis()
        
        val requestTime = measureTimeMillis {
            // 空操作，仅用于测时
        }
        
        val response = chain.proceed(request)
        val responseTime = System.currentTimeMillis() - startTime
        
        val metrics = RequestMetrics(
            url = request.url.toString(),
            method = request.method,
            statusCode = response.code,
            requestTime = requestTime,
            responseTime = responseTime,
            totalTime = System.currentTimeMillis() - startTime,
            requestSize = request.body?.contentLength() ?: 0,
            responseSize = response.body?.contentLength() ?: 0,
            isFromCache = response.header("X-From-Cache") != null
        )
        
        // 上报性能数据
        reportMetrics(metrics)
        
        return response
    }

            append("Request: ${metrics.requestTime}ms\n")

            append("Total: ${metrics.totalTime}ms\n")
            append("Request Size: ${metrics.requestSize} bytes\n")
```

```kotlin
fun <T> Observable<T>.transformNetworkError(): Observable<T> {
    return this.onErrorResumeNext { throwable ->
        Observable.error(

                is HttpException -> {

                            401,

                        else -> NetworkException.HttpException(
                            throwable.code(),
                            "HTTP ${throwable.code()}: ${throwable.message}"
                        )
                    }
                }
                is SocketTimeoutException -> NetworkException.TimeoutException()
                is ConnectException -> NetworkException.NetworkError("无法连接到服务器")
                is IOException -> NetworkException.NetworkError("网络连接错误")
                is JsonSyntaxException -> NetworkException.ParseException("JSON解析失败")
                else -> NetworkException.NetworkError(
                    throwable.message ?: "未知错误"
                )
            }
        )
    }
}
```

### 2. ViewModel中的错误处理

```kotlin
class UserListViewModelWithErrorHandling @Inject constructor(
    private val repository: UserRepository
) : ViewModel() {
    
    private val _uiState = MutableStateFlow<UiState<List<User>>>(UiState.Idle)
    val uiState = _uiState.asStateFlow()
    
    private val _error = MutableSharedFlow<ErrorUIModel>()
    val error = _error.asSharedFlow()
    
    private val disposables = CompositeDisposable()
    
    fun loadUsers(page: Int = 1) {
        repository.getUserList(page)
            .transformNetworkError()
            .doOnSubscribe { _uiState.value = UiState.Loading }
            .subscribe(
                { users ->
                    _uiState.value = UiState.Success(users)
                },
                { error ->
                    handleError(error as NetworkException)
                }
            )
            .let { disposables.add(it) }
    }
    
    private fun handleError(error: NetworkException) {
        _uiState.value = UiState.Error(error.userMessage)
        
        viewModelScope.launch {
            _error.emit(
                ErrorUIModel(
                    message = error.userMessage,
                    actionText = when (error) {
                        is NetworkException.TimeoutException -> "重试"
                        is NetworkException.NetworkError -> "重试"
                        else -> "关闭"
                    },
                    action = when (error) {
                        is NetworkException.TimeoutException -> { loadUsers() }
                        is NetworkException.NetworkError -> { loadUsers() }
                        else -> {}
                    }
                )
            )
        }
    }
    
    override fun onCleared() {
        super.onCleared()
        disposables.dispose()
    }
}

data class ErrorUIModel(
    val message: String,
    val actionText: String,
    val action: () -> Unit
)
```

## 最佳实践总结

|场景|推荐方案|
|---|---|
|列表数据|缓存优先 + 后台更新|
|详情数据|网络优先 + 降级缓存|
|用户操作|离线队列 + 网络恢复同步|
|高并发|限流 + 并发度控制|
|防止卡顿|连接池 + 线程调度优化|
|错误恢复|指数退避重试 + 降级策略|
