```kotlin
                        // 错误处理
                    }

            val articlesLiveData = repository.fetchArticles(page, PAGE_SIZE)
            addSource(articlesLiveData) { resource ->
                value = resource is Resource.Loading
            }
        }
    }
    
    fun loadNextPage() {
        val nextPage = (_currentPage.value ?: 0) + 1
        _currentPage.value = nextPage
    }
```

```kotlin
// SearchViewModel.kt（RxJava版）
class SearchViewModel(private val repository: SearchRepository) : ViewModel() {
    private val searchQuerySubject = BehaviorSubject.createDefault("")
    private val compositeDisposable = CompositeDisposable()
    
    val searchResults: Observable<List<SearchResult>> = searchQuerySubject
        .debounce(500, TimeUnit.MILLISECONDS)  // 防抖
        .distinctUntilChanged()                 // 去重
        .switchMap { query ->
            if (query.isEmpty()) {
                Observable.just(emptyList())
            } else {
                repository.search(query)
                    .subscribeOn(Schedulers.io())
                    .observeOn(AndroidSchedulers.mainThread())
            }
        }
        .startWith(emptyList())
    
    val isSearching: Observable<Boolean> = searchQuerySubject
        .debounce(500, TimeUnit.MILLISECONDS)
        .map { it.isNotEmpty() }
        .startWith(false)
    
    fun onSearchQueryChanged(query: String) {
        searchQuerySubject.onNext(query)
    }
    
    override fun onCleared() {
        compositeDisposable.clear()
    }
}
```

```kotlin
// FormViewModel.kt（LiveData版）
class FormViewModel : ViewModel() {
    private val emailField = MutableLiveData<String>("")
    private val passwordField = MutableLiveData<String>("")
    private val confirmPasswordField = MutableLiveData<String>("")
    
    // 邮箱验证
    val emailError: LiveData<String?> = Transformations.map(emailField) { email ->
        when {
            email.isEmpty() -> "邮箱不能为空"
            !email.contains("@") -> "邮箱格式不正确"
            else -> null
        }
    }
    
    // 密码验证
    val passwordError: LiveData<String?> = Transformations.map(passwordField) { password ->
        when {
            password.isEmpty() -> "密码不能为空"
            password.length < 8 -> "密码至少8位"
            else -> null
        }
    }
    
    // 表单是否有效（复杂的 MediatorLiveData）
    val formValid: LiveData<Boolean> = MediatorLiveData<Boolean>().apply {
        var currentEmail = ""
        var currentPassword = ""
        var currentConfirmPassword = ""
        
        addSource(emailField) { email ->
            currentEmail = email
            value = isFormValid(currentEmail, currentPassword, currentConfirmPassword)
        }
        addSource(passwordField) { password ->
            currentPassword = password
            value = isFormValid(currentEmail, currentPassword, currentConfirmPassword)
        }
        addSource(confirmPasswordField) { confirmPassword ->
            currentConfirmPassword = confirmPassword
            value = isFormValid(currentEmail, currentPassword, currentConfirmPassword)
        }
    }
    
    private fun isFormValid(email: String, password: String, confirmPassword: String): Boolean {
        return email.isNotEmpty() && email.contains("@") &&
               password.isNotEmpty() && password.length >= 8 &&
               confirmPassword.isNotEmpty() && password == confirmPassword
    }
    
    fun setEmail(email: String) { emailField.value = email }
    fun setPassword(password: String) { passwordField.value = password }
    fun setConfirmPassword(confirmPassword: String) { confirmPasswordField.value = confirmPassword }
}
```

```kotlin
class DashboardViewModel(
    private val userRepository: UserRepository,
    private val analyticsRepository: AnalyticsRepository,
    private val notificationRepository: NotificationRepository
) : ViewModel() {
    
    private val _userId = MutableStateFlow<String>("")
    
    // 合并多个数据源
    val dashboardData: StateFlow<DashboardState> = _userId
        .flatMapLatest { userId ->
            if (userId.isEmpty()) {
                flowOf(DashboardState.Empty)
            } else {
                combine(
                    userRepository.getUser(userId),
                    analyticsRepository.getAnalytics(userId),
                    notificationRepository.getNotifications(userId)
                ) { user, analytics, notifications ->
                    DashboardState.Success(
                        user = user,
                        analytics = analytics,
                        notifications = notifications
                    )
                }
            }
        }
        .catch { exception ->
            emit(DashboardState.Error(exception))
        }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), DashboardState.Loading)
    
    fun loadDashboard(userId: String) {
        _userId.value = userId
    }
}

    object Empty : DashboardState()
    data class Success(
        val user: User,
        val analytics: Analytics,
        val notifications: List<Notification>
```

```kotlin
// ==================== Model 层 ====================
// api/UserApi.kt
interface UserApi {
    @GET("/users")
    suspend fun getUsers(@Query("page") page: Int): List<User>
    
    @GET("/users/{id}")
    suspend fun getUserDetail(@Path("id") userId: String): User
    
    @GET("/users/search")
    suspend fun searchUsers(@Query("q") query: String): List<User>
}

// entity/User.kt
data class User(
    val id: String,
    val name: String,
    val avatar: String,
    val description: String
)

// repository/UserRepository.kt
class UserRepository(
    private val api: UserApi,
    private val database: UserDao
) {
    // 获取用户列表（Flow）
    fun getUserList(page: Int): Flow<Resource<List<User>>> = flow {
        emit(Resource.Loading())
        try {
            val users = api.getUsers(page)
            database.insertUsers(users)
            emit(Resource.Success(users))
        } catch (e: Exception) {
            val cached = database.getAllUsers()
            if (cached.isNotEmpty()) {
                emit(Resource.Success(cached))
            } else {
                emit(Resource.Error(e))
            }
        }
    }.flowOn(Dispatchers.IO)
    
    // 获取用户详情（Flow）
    fun getUserDetail(userId: String): Flow<Resource<User>> = flow {
        emit(Resource.Loading())
        try {
            val user = api.getUserDetail(userId)
            database.insertUser(user)
            emit(Resource.Success(user))
        } catch (e: Exception) {
            val cached = database.getUser(userId)
            if (cached != null) {
                emit(Resource.Success(cached))
            } else {
                emit(Resource.Error(e))
            }
        }
    }.flowOn(Dispatchers.IO)
    
    // 搜索用户（Flow）
    fun searchUsers(query: String): Flow<Resource<List<User>>> = flow {
        emit(Resource.Loading())
        try {
            val results = api.searchUsers(query)
            emit(Resource.Success(results))
        } catch (e: Exception) {
            emit(Resource.Error(e))
        }
    }.flowOn(Dispatchers.IO)
}

// ==================== ViewModel 层 ====================
// viewmodel/UserListViewModel.kt
@HiltViewModel
class UserListViewModel @Inject constructor(
    private val repository: UserRepository
) : ViewModel() {
    
    // 页码状态
    private val _currentPage = MutableStateFlow(1)
    
    // 搜索查询状态
    private val _searchQuery = MutableStateFlow("")
    
    // 用户列表状态
    val userList: StateFlow<Resource<List<User>>> = combine(
        _currentPage,
        _searchQuery
    ) { page, query ->
        if (query.isEmpty()) page to false else page to true
    }
        .flatMapLatest { (page, isSearching) ->
            if (isSearching) {
                repository.searchUsers(_searchQuery.value)
            } else {
                repository.getUserList(page)
            }
        }
        .stateIn(
            viewModelScope,
            SharingStarted.WhileSubscribed(5000),
            Resource.Loading()
        )
    
    // 分页指示器
    val isPaging: StateFlow<Boolean> = combine(
        _currentPage,
        userList
    ) { page, resource ->
        page > 1 && resource is Resource.Loading
    }.stateIn(viewModelScope, SharingStarted.Lazily, false)
    
    fun searchUsers(query: String) {
        _searchQuery.value = query
        _currentPage.value = 1  // 搜索时重置页码
    }
    
    fun loadNextPage() {
        if (_searchQuery.value.isEmpty()) {
            _currentPage.value += 1
        }
    }
    
    fun refresh() {
        _currentPage.value = 1
        _searchQuery.value = ""
    }
}
```

## MVI架构实现

### MVI 框架解析（Model-View-Intent）

**MVI 核心思想：**

```
User Intent (用户行为)
    ↓
Action (转化为操作)
    ↓
Reducer (更新状态)
    ↓
State (当前状态)
    ↓
Render (渲染UI)
```

```kotlin
// ==================== MVI Models ====================
// mvi/UserListViewState.kt
data class UserListViewState(
    val users: List<User> = emptyList(),
    val isLoading: Boolean = false,
    val error: Throwable? = null,
    val searchQuery: String = "",
    val currentPage: Int = 1
)

// mvi/UserListViewIntent.kt
sealed class UserListViewIntent {
    object LoadUsers : UserListViewIntent()
    object LoadNextPage : UserListViewIntent()
    data class SearchUsers(val query: String) : UserListViewIntent()
    object Refresh : UserListViewIntent()
}

// mvi/UserListAction.kt
sealed class UserListAction {
    object LoadUsersAction : UserListAction()
    object LoadNextPageAction : UserListAction()
    data class SearchAction(val query: String) : UserListAction()
    object RefreshAction : UserListAction()
}

// mvi/UserListReducer.kt
object UserListReducer {
    fun reduce(
        currentState: UserListViewState,
        action: UserListAction
    ): UserListViewState = when (action) {
        is UserListAction.LoadUsersAction -> 
            currentState.copy(isLoading = true, error = null)
        
        is UserListAction.SearchAction -> 
            currentState.copy(
                searchQuery = action.query,
                currentPage = 1,
                isLoading = true
            )
        
        is UserListAction.RefreshAction -> 
            currentState.copy(currentPage = 1, isLoading = true)
        
        is UserListAction.LoadNextPageAction -> 
            currentState.copy(currentPage = currentState.currentPage + 1)
    }
}

// ==================== MVI ViewModel ====================
// viewmodel/UserListMVIViewModel.kt
@HiltViewModel
class UserListMVIViewModel @Inject constructor(
    private val repository: UserRepository
) : ViewModel() {
    
    // Intent 流 - 接收用户操作
    private val intentFlow = MutableSharedFlow<UserListViewIntent>()
    
    // State 流 - 当前状态
    private val _viewState = MutableStateFlow(UserListViewState())
    val viewState: StateFlow<UserListViewState> = _viewState.asStateFlow()
    
    init {
        // 处理 Intent -> Action -> State
        viewModelScope.launch {
            intentFlow
                .map { intent -> intentToAction(intent) }
                .flatMapLatest { action -> actionToFlow(action) }
                .collect { action ->
                    val currentState = _viewState.value
                    val newState = UserListReducer.reduce(currentState, action)
                    _viewState.value = newState
                }
        }
    }
    
    // Intent 转 Action
    private fun intentToAction(intent: UserListViewIntent): UserListAction = when (intent) {
        UserListViewIntent.LoadUsers -> UserListAction.LoadUsersAction
        UserListViewIntent.LoadNextPage -> UserListAction.LoadNextPageAction
        is UserListViewIntent.SearchUsers -> UserListAction.SearchAction(intent.query)
        UserListViewIntent.Refresh -> UserListAction.RefreshAction
    }
    
    // Action 转 Data Flow
    private fun actionToFlow(action: UserListAction): Flow<UserListAction> = when (action) {
        UserListAction.LoadUsersAction -> {
            flow {
                val state = _viewState.value
                repository.getUserList(state.currentPage)
                    .catch { exception ->
                        _viewState.update { it.copy(error = exception, isLoading = false) }
                    }
                    .collect { resource ->
                        when (resource) {
                            is Resource.Success -> {
                                _viewState.update { 
                                    it.copy(
                                        users = resource.data,
                                        isLoading = false
                                    ) 
                                }
                            }
                            is Resource.Error -> {
                                _viewState.update { 
                                    it.copy(error = resource.exception, isLoading = false) 
                                }
                            }
                            else -> {}
                        }
                    }
            }
        }
        
        is UserListAction.SearchAction -> {
            flow {
                repository.searchUsers(action.query)
                    .catch { exception ->
                        _viewState.update { it.copy(error = exception, isLoading = false) }
                    }
                    .collect { resource ->
                        when (resource) {
                            is Resource.Success -> {
                                _viewState.update { 
                                    it.copy(users = resource.data, isLoading = false) 
                                }
                            }
                            else -> {}
                        }
                    }
            }
        }
        
        else -> flowOf(action)
    }
    
    // Public methods - UI 调用
    fun processIntent(intent: UserListViewIntent) {
        viewModelScope.launch {
            intentFlow.emit(intent)
        }
    }
    
    fun loadUsers() = processIntent(UserListViewIntent.LoadUsers)
    fun search(query: String) = processIntent(UserListViewIntent.SearchUsers(query))
    fun loadNextPage() = processIntent(UserListViewIntent.LoadNextPage)
    fun refresh() = processIntent(UserListViewIntent.Refresh)
}

// ==================== MVI Fragment ====================
// fragment/UserListMVIFragment.kt
@AndroidEntryPoint
class UserListMVIFragment : Fragment() {
    private val viewModel: UserListMVIViewModel by viewModels()
    private lateinit var adapter: UserAdapter
    
    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)
        
        adapter = UserAdapter { user ->
            navigateToDetail(user.id)
        }
        
        recyclerView.adapter = adapter
        
        // 监听状态变化 - 单一数据源
        lifecycleScope.launchWhenStarted {
            viewModel.viewState.collect { state ->
                render(state)
            }
        }
        
        // 处理用户 Intent
        searchEditText.addTextChangedListener { text ->
            viewModel.search(text?.toString() ?: "")
        }
        
        swipeRefreshLayout.setOnRefreshListener {
            viewModel.refresh()
        }
    }
    
    // 单一的 render 函数 - MVI 的核心特性
    private fun render(state: UserListViewState) {
        // 渲染用户列表
        adapter.submitList(state.users)
        
        // 渲染加载状态
        progressBar.visibility = if (state.isLoading && state.currentPage == 1) {
            View.VISIBLE
        } else {
            View.GONE
        }
        
        // 渲染分页加载状态
        paginationProgressBar.visibility = if (state.isLoading && state.currentPage > 1) {
            View.VISIBLE
        } else {
            View.GONE
        }
        
        swipeRefreshLayout.isRefreshing = false
        
        // 渲染错误状态
        if (state.error != null) {
            showSnackbar(state.error.message ?: "加载失败")
        }
        
        // 渲染搜索状态
        if (state.searchQuery.isNotEmpty()) {
            searchEditText.setText(state.searchQuery)
        }
    }
}
```

```kotlin
    fun setUserId(userId: String) {

    }
}

        lifecycleScope.launchWhenStarted {

                showUserInfo(user)

        // 3. 观察分析事件（RxJava）

            }
    }
    
    override fun onDestroyView() {

    }
}
```

## 不能混用的情况

### 情况1：在 ViewModel 中同时返回 LiveData 和 Flow（✗ 不要做）

```kotlin
// ❌ 错误示范
@HiltViewModel

    val userLiveData: LiveData<User> = repository.getUserLiveData()
    val userFlow: StateFlow<User> = repository.getUserFlow()
    
    // 问题1：两个数据源可能不同步
    // 问题2：难以维护，观察者会困惑应该用哪个
    // 问题3：增加包体积和复杂度
}

// ✓ 正确做法：统一接口
@HiltViewModel
class GoodViewModel(repository: UserRepository) : ViewModel() {
    val user: StateFlow<User> = repository.getUserFlow()
        .stateIn(viewModelScope, SharingStarted.Lazily, User())
    
    // 如果必须兼容 LiveData（旧代码），提供适配：
    val userLiveData: LiveData<User> = user.asLiveData()
}
```

### 情况2：Fragment 中订阅后不释放（✗ 内存泄漏风险）

```kotlin
// ❌ 错误示范 - RxJava 内存泄漏
class BadFragment : Fragment() {
    private val viewModel: UserViewModel by viewModels()

        compositeDisposable.dispose()
    }
}
```

## 总结建议表

|选择|场景|优点|缺点|
|---|---|---|---|
|**LiveData**|UI 状态管理|自动生命周期、包小|操作符少|
|**Flow**|现代 Jetpack|协程原生、背压自动|学习曲线|
|**RxJava**|复杂业务|操作符多、成熟|包大、需手动管理|
|**混用**|迁移过程|平滑过渡|复杂度高、难维护|

**最佳实践：**

```
新项目    -> 100% Flow + StateFlow
迁移项目  -> Flow + Repository层RxJava + 适配
旧项目    -> 保持RxJava（除非必要）
```
