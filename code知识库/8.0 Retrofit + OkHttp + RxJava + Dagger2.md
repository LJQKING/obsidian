class LoggingInterceptor : Interceptor {
    companion object {
        private const val TAG = "OkHttp"
    }

    override fun intercept(chain: Interceptor.Chain): Response {
        val request = chain.request()
        val startTime = System.currentTimeMillis()

        val response = try {
            chain.proceed(request)
        } catch (e: Exception) {
            Log.e(TAG, "Request failed: ${request.url}", e)
            throw e
        }

        val endTime = System.currentTimeMillis()
        val duration = endTime - startTime

        Log.d(TAG, buildString {
            append("\n")

// 2. 请求头公共参数拦截器
class HeaderInterceptor @Inject constructor(
    private val context: Context
) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val original = chain.request()
        
        val request = original.newBuilder()
            .header("Content-Type", "application/json")
            .header("Accept", "application/json")
            .header("User-Agent", "MyApp/${BuildConfig.VERSION_NAME}")
            .header("X-Timestamp", System.currentTimeMillis().toString())
            .header("X-Device-Id", getDeviceId(context))
            .header("Authorization", "Bearer ${getAuthToken(context)}")
            .build()

        return chain.proceed(request)
    }

    private fun getDeviceId(context: Context): String {
        return Settings.Secure.getString(
            context.contentResolver,
            Settings.Secure.ANDROID_ID
        )
    }

    private fun getAuthToken(context: Context): String {
        // 从SharedPreferences或数据库获取token
        val sharedPref = context.getSharedPreferences("auth", Context.MODE_PRIVATE)
        return sharedPref.getString("token", "") ?: ""
    }
}

// 3. 重试与超时拦截器
class RetryInterceptor(
    private val maxRetry: Int = 3,
    private val retryDelayMs: Long = 1000
) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        var request = chain.request()
        var exception: Exception? = null

        for (attempt in 1..maxRetry) {
            try {
                return chain.proceed(request)
            } catch (e: Exception) {
                exception = e
                Log.w("RetryInterceptor", "Attempt $attempt failed, retrying...", e)
                
                // 指数退避：第1次1秒，第2次2秒，第3次4秒
                val delayMs = retryDelayMs * (1 shl (attempt - 1))
                Thread.sleep(delayMs)
            }
        }

        throw exception ?: Exception("Max retries exceeded")
    }
}

// 4. 响应拦截器 - 处理错误和token刷新
class ResponseInterceptor @Inject constructor(
    private val context: Context
) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val response = chain.proceed(chain.request())

        // 处理401 Unauthorized - Token过期
        if (response.code == 401) {
            // 尝试刷新token
            val refreshed = refreshToken()
            if (refreshed) {
                // 使用新token重试原请求
                val newRequest = chain.request()
                    .newBuilder()
                    .header("Authorization", "Bearer ${getNewToken(context)}")
                    .build()
                return chain.proceed(newRequest)
            }
        }

        return response
    }

    private fun refreshToken(): Boolean {
        // 调用刷新token API
        return true
    }

    private fun getNewToken(context: Context): String {
        val sharedPref = context.getSharedPreferences("auth", Context.MODE_PRIVATE)
        return sharedPref.getString("token", "") ?: ""
    }
}

// 1. RxJava操作符集合工具
object RxJavaUtils {
    
    /**
     * 统一处理API响应和错误
     * 如果code != 200则转换为错误
     */
    fun <T> handleApiResponse(): ObservableTransformer<ApiResponse<T>, T> {
        return ObservableTransformer { upstream ->
            upstream
                .flatMap { response ->
                    if (response.isSuccess() && response.data != null) {
                        Observable.just(response.data)
                    } else {
                        Observable.error(
                            ApiException(
                                code = response.code,
                                message = response.message
                            )
                        )
                    }
                }
                .onErrorResumeNext { throwable ->
                    Observable.error(handleException(throwable))
                }
        }
    }

    /**
     * 线程调度：IO线程请求，Main线程回调
     */
    fun <T> ioToMain(): ObservableTransformer<T, T> {
        return ObservableTransformer { upstream ->
            upstream
                .subscribeOn(Schedulers.io())
                .observeOn(AndroidSchedulers.mainThread())
        }
    }

    /**
     * 超时处理
     */
    fun <T> withTimeout(timeoutSeconds: Long = 30): ObservableTransformer<T, T> {
        return ObservableTransformer { upstream ->
            upstream.timeout(timeoutSeconds, TimeUnit.SECONDS)
        }
    }

    /**
     * 重试逻辑：指数退避
     */
    fun <T> retryWithExponentialBackoff(
        maxRetry: Int = 3,
        delayMs: Long = 1000
    ): ObservableTransformer<T, T> {
        return ObservableTransformer { upstream ->
            upstream.retryWhen { errors ->
                errors.zipWith(
                    Observable.range(1, maxRetry),
                    BiFunction { error: Throwable, attemptNumber: Int ->
                        if (attemptNumber > maxRetry) {
                            throw error
                        }
                        // 指数退避延迟
                        val delay = delayMs * (1L shl (attemptNumber - 1))
                        delay
                    }
                )
                    .flatMap { delay ->
                        Observable.timer(delay, TimeUnit.MILLISECONDS)
                    }
            }
        }
    }

    /**
     * 加载状态管理
     */
    fun <T> withLoadingState(
        onShowLoading: () -> Unit,
        onHideLoading: () -> Unit
    ): ObservableTransformer<T, T> {
        return ObservableTransformer { upstream ->
            upstream
                .doOnSubscribe { onShowLoading() }
                .doOnNext { onHideLoading() }
                .doOnError { onHideLoading() }
                .doOnComplete { onHideLoading() }
        }
    }

    /**
     * 统一异常处理
     */
    private fun handleException(throwable: Throwable): Throwable {
        return when (throwable) {
            is HttpException -> {
                // HTTP错误
                ApiException(
                    code = throwable.code(),
                    message = "HTTP ${throwable.code()}: ${throwable.message}"
                )
            }
            is SocketTimeoutException -> {
                ApiException(
                    code = -1001,
                    message = "连接超时，请检查网络"
                )
            }
            is ConnectException -> {
                ApiException(
                    code = -1002,
                    message = "网络连接失败"
                )
            }
            is IOException -> {
                ApiException(
                    code = -1003,
                    message = "网络IO错误"
                )
            }
            else -> {
                ApiException(
                    code = -1004,
                    message = throwable.message ?: "未知错误"
                )
            }
        }
    }
}

class UserRepository @Inject constructor(
    private val apiService: UserApiService

    /**
     * 获取用户列表 - 完整链路演示
     */
    fun getUserList(page: Int, pageSize: Int): Observable<UserListResponse> {
        return apiService.getUserList(page, pageSize)
            // 1. 处理API响应格式
            .compose(RxJavaUtils.handleApiResponse())
            
            // 2. 添加超时处理
            .compose(RxJavaUtils.withTimeout(30))
            
            // 3. 添加重试逻辑（仅在网络错误时）
            .compose(RxJavaUtils.retryWithExponentialBackoff(maxRetry = 2))
            
            // 4. 线程切换：IO请求，Main回调
            .compose(RxJavaUtils.ioToMain())
    }

    /**
     * 获取用户详情
     */
    fun getUserDetail(userId: Long): Observable<User> {
        return apiService.getUserDetail(userId)
            .compose(RxJavaUtils.handleApiResponse())
            .compose(RxJavaUtils.withTimeout(20))
            .compose(RxJavaUtils.ioToMain())
    }

    /**
     * 创建用户
     */
    fun createUser(user: User): Observable<User> {
        return apiService.createUser(user)
            .compose(RxJavaUtils.handleApiResponse())
            .compose(RxJavaUtils.ioToMain())
    }

    /**
     * 批量获取用户 - 演示合并多个请求
     */
    fun getUsersInBatch(userIds: List<Long>): Observable<List<User>> {
        return Observable.fromIterable(userIds)
            .flatMapSingle { userId ->
                apiService.getUserDetail(userId)
                    .compose(RxJavaUtils.handleApiResponse())
                    .firstOrError()
            }
            .toList()
            .toObservable()
            .compose(RxJavaUtils.ioToMain())
    }

    /**
     * 并行请求多个用户 - 演示zip操作
     */
    fun getUsersParallel(userId1: Long, userId2: Long): Observable<Pair<User, User>> {
        return Observable.zip(
            apiService.getUserDetail(userId1)
                .compose(RxJavaUtils.handleApiResponse()),
            apiService.getUserDetail(userId2)
                .compose(RxJavaUtils.handleApiResponse()),
            BiFunction { user1: User, user2: User ->
                Pair(user1, user2)
            }
        )
            .compose(RxJavaUtils.ioToMain())
    }
}

## 十二、常见问题解答

### Q1: 如何处理Token过期？

**A:** ResponseInterceptor中检测401状态，自动调用刷新接口更新token，然后重试原请求。

### Q2: 如何优雅地处理多个请求的并发？

**A:** 使用RxJava的`zip()`操作符并行请求，`flatMap()`串行请求，`concat()`按顺序请求。

### Q3: 为何使用RxJava而不是Coroutine？

**A:** 两者都可用。RxJava操作符丰富但学习陡峭，Coroutine更简洁但操作符少。可混用。

### Q4: 如何测试网络层？

**A:** 使用MockWebServer模拟HTTP服务，配合JUnit4和PowerMock进行单元测试。

### Q5: 生产环境如何隐藏日志？

**A:** 通过BuildConfig.DEBUG判断，发布版本不添加LoggingInterceptor。
