
## 模块一：Android 基础 IPC 体系

### 1.1 四大组件与生命周期

#### Activity 生命周期与任务栈

```
onCreate → onStart → onResume → (running) → onPause → onStop → onDestroy

关键状态转换：
- onResume: 获得焦点，获取资源（音频焦点、传感器等）
- onPause: 失去焦点，释放资源（暂停音频）
- onDestroy: 清理（取消监听、释放引用）
```

**智能座舱场景**：

- Activity 位于不同的虚拟显示器（IVI 屏幕 vs 仪表盘）
- 需要处理 multi-display awareness
- 音频焦点管理与 Activity 生命周期关联

#### Service 四大类型

|类型|进程|生命周期|应用场景|
|---|---|---|---|
|**Started Service**|后台进程|startService → onStartCommand → stopService|音频播放、数据同步|
|**Bound Service**|同进程/IPC|bindService → onBind → unbindService|音乐播放器服务、语音识别服务|
|**Foreground Service**|前台进程|startForegroundService + setForeground|语音识别（ASR）持续运行|
|**JobScheduler**|后台|系统调度|定时任务、网络同步|

**源码实现细节**：

```kotlin
// 前台服务（语音识别场景）

            MediaRecorder.AudioSource.MIC,
            SAMPLE_RATE,
```

- `START_STICKY` vs `START_NOT_STICKY` vs `START_REDELIVER_INTENT` 的区别
- Foreground Service 通知必须存在，否则 ANR
- 进程优先级：Foreground (100) > Visible (200) > Service (300) > Background (400) > Empty (500)

### 1.2 Binder 与 AIDL 深度分析

#### 进程间通信的本质

```
User Space (App Process)           Kernel Space              User Space (Service Process)
┌─────────────────────────────┐    ┌──────────┐            ┌─────────────────────────────┐
│ IBinder (Proxy)             │────│ Binder   │────────────│ BBinder (Service)           │
│ transact(code, data) ──────────► Driver   │ ◄──────────│ onTransact(code, data)      │
│ reply = parcel receive      │◄───│         │ ────────────►│ return reply parcel         │
└─────────────────────────────┘    └──────────┘            └─────────────────────────────┘
```

#### AIDL 编译生成的代码结构

假设定义：

```aidl
// IVoiceRecognitionService.aidl
package com.example.voice;

interface IVoiceRecognitionService {
    void startListening();
    String recognizeAudio(in byte[] audioData);
}
```

**编译生成的代码**：

```java
public interface IVoiceRecognitionService extends android.os.IInterface {
    // Stub 类：服务端实现
    public static abstract class Stub extends android.os.Binder 
            implements IVoiceRecognitionService {
        
        private static final String DESCRIPTOR = 
            "com.example.voice.IVoiceRecognitionService";
        
        public Stub() {
            this.attachInterface(this, DESCRIPTOR);
        }
        
        // 获取代理对象
        public static IVoiceRecognitionService asInterface(android.os.IBinder obj) {
            if ((obj == null)) {
                return null;
            }
            IInterface iin = obj.queryLocalInterface(DESCRIPTOR);
            if (((iin) != null) && (iin instanceof IVoiceRecognitionService)) {
                return ((IVoiceRecognitionService) iin);
            }
            return new Proxy(obj);  // 创建代理对象
        }
        
        @Override
        public boolean onTransact(int code, Parcel data, Parcel reply, 
                int flags) throws RemoteException {
            String descriptor = DESCRIPTOR;
            
            switch (code) {
                case INTERFACE_TRANSACTION: {
                    reply.writeString(descriptor);
                    return true;
                }
                case TRANSACTION_startListening: {
                    data.enforceInterface(descriptor);
                    this.startListening();  // 调用实现方法
                    reply.writeNoException();
                    return true;
                }
                case TRANSACTION_recognizeAudio: {
                    data.enforceInterface(descriptor);
                    byte[] _arg0 = data.createByteArray();
                    String _result = this.recognizeAudio(_arg0);
                    reply.writeNoException();
                    reply.writeString(_result);
                    return true;
                }
                default: {
                    return super.onTransact(code, data, reply, flags);
                }
            }
        }
    }
    
    // Proxy 类：客户端代理对象
    public static class Proxy implements IVoiceRecognitionService {
        private android.os.IBinder mRemote;
        
        Proxy(android.os.IBinder remote) {
            mRemote = remote;
        }
        
        @Override
        public void startListening() throws RemoteException {
            Parcel _data = Parcel.obtain();
            Parcel _reply = Parcel.obtain();
            try {
                _data.writeInterfaceToken(DESCRIPTOR);
                mRemote.transact(Stub.TRANSACTION_startListening, 
                    _data, _reply, 0);
                _reply.readException();
            } finally {
                _reply.recycle();
                _data.recycle();
            }
        }
        
        @Override
        public String recognizeAudio(byte[] audioData) throws RemoteException {
            Parcel _data = Parcel.obtain();
            Parcel _reply = Parcel.obtain();
            String _result;
            try {
                _data.writeInterfaceToken(DESCRIPTOR);
                _data.writeByteArray(audioData);
                mRemote.transact(Stub.TRANSACTION_recognizeAudio, 
                    _data, _reply, 0);
                _reply.readException();
                _result = _reply.readString();
            } finally {
                _reply.recycle();
                _data.recycle();
            }
            return _result;
        }
    }
}
```

**关键概念**：

- **IBinder**：Binder 驱动的客户端抽象
- **Parcel**：数据序列化容器（类似 Protocol Buffers）
- **DESCRIPTOR**：接口唯一标识
- **transact()**：跨进程调用的核心方法

**性能陷阱**：

```kotlin
// ❌ 错误：同步 IPC 阻塞 UI 线程
val result = voiceService.recognizeAudio(audioBytes)  // 可能阻塞数秒

// ✅ 正确：异步处理
lifecycleScope.launch(Dispatchers.Default) {
    val result = voiceService.recognizeAudio(audioBytes)
    withContext(Dispatchers.Main) {
        updateUI(result)
    }
}

// ✅ 更优：使用回调或 Flow
voiceService.recognizeAudioAsync(audioBytes) { result ->
    runOnUiThread { updateUI(result) }
}
```

### 1.3 Handler 与 MessageQueue 源码剖析

#### 线程通信模型

```
Looper (单线程关联)
    ↓
MessageQueue (FIFO 队列)
    ↓
Handler (消息发送者/处理器)
    ↓
Message (数据载体)
```

**源码流程**：

```kotlin
// 1. 创建 Handler（必须在有 Looper 的线程中）
class VoiceProcessingThread : Thread() {
    lateinit var voiceHandler: Handler
    
    override fun run() {
        Looper.prepare()  // 为线程创建 Looper
        voiceHandler = Handler(Looper.myLooper()!!) { msg ->
            when (msg.what) {
                MSG_START_RECOGNITION -> startRecognition()
                MSG_PROCESS_AUDIO -> processAudio(msg.obj as ByteArray)
            }
            true
        }
        Looper.loop()  // 开始循环处理消息
    }
}

// 2. 从主线程发送消息
voiceHandler.post {
    // 运行在 VoiceProcessingThread 中
    startAudioRecording()
}

// 3. 发送延迟消息
voiceHandler.postDelayed({
    stopAudioRecording()
}, 5000)  // 5秒后停止录音

// 4. 消息优先级和屏障
voiceHandler.sendMessageAtFrontOfQueue(Message.obtain().apply {
    what = MSG_URGENT_STOP
})
```

**MessageQueue 内部实现**：

```java
// 简化版源码
public class MessageQueue {
    private Message mMessages;  // 链表头
    
    boolean enqueueMessage(Message msg, long when) {
        synchronized (this) {
            msg.when = when;
            Message p = mMessages;
            boolean needWake;
            
            // 按时间戳排序插入
            if (p == null || when == 0 || when < p.when) {
                msg.next = p;
                mMessages = msg;
                needWake = mBlocked;
            } else {
                Message prev;
                while (p != null && p.when <= when) {
                    prev = p;
                    p = p.next;
                }
                msg.next = p;
                prev.next = msg;
                needWake = false;
            }
            
            if (needWake) {
                nativeWake(mPtr);  // 唤醒 poll()
            }
        }
        return true;
    }
    
    Message next() {
        int nextPollTimeoutMillis = 0;
        for (;;) {
            nativePollOnce(mPtr, nextPollTimeoutMillis);  // 等待事件
            
            synchronized (this) {
                final long now = SystemClock.uptimeMillis();
                Message prevMsg = null;
                Message msg = mMessages;
                
                if (msg != null && msg.when <= now) {
                    // 消息已准备好
                    mMessages = msg.next;
                    msg.next = null;
                    return msg;
                } else if (msg != null) {
                    // 计算等待时间
                    nextPollTimeoutMillis = 
                        (int) Math.min(msg.when - now, Integer.MAX_VALUE);
                }
            }
        }
    }
}
```

**音频处理中的应用**：

```kotlin
// 实时音频数据处理线程
class AudioProcessingThread : Thread() {
    private lateinit var audioHandler: Handler
    
    override fun run() {
        Looper.prepare()
        audioHandler = object : Handler(Looper.myLooper()!!) {
            override fun handleMessage(msg: Message) {
                when (msg.what) {
                    MSG_AUDIO_FRAME -> {
                        val audioFrame = msg.obj as AudioFrame
                        // 执行 DSP 处理（低延迟）
                        processAudioFrame(audioFrame)
                        // 立即投递下一帧消息（避免队列堆积）
                        sendMessage(obtainMessage(MSG_AUDIO_FRAME, nextFrame))
                    }
                }
            }
        }
        Looper.loop()
    }
}

// 注意：对于实时音频，AudioRecord 通常运行在专用线程，
// 不要依赖 Handler 处理每一帧音频（延迟太高）
```

### 1.4 多线程与 RxJava vs Kotlin Coroutines 在 AAOS 中的选择

#### RxJava 2/3 架构

```kotlin
// 语音识别流程的 RxJava 实现
class VoiceRecognitionViewModel(
    private val voiceService: VoiceRecognitionService
) {
    private val disposable = CompositeDisposable()
    
    fun startRecognition(): Observable<RecognitionResult> {
        return voiceService.recordAudio()  // Observable<ByteArray>
            .flatMap { audioBytes ->
                // 发送到 ASR 服务
                voiceService.sendToASR(audioBytes)
            }
            .map { asrResult ->
                // 解析 ASR 结果
                parseASRResult(asrResult)
            }
            .subscribeOn(Schedulers.io())  // 订阅线程（I/O线程）
            .observeOn(AndroidSchedulers.mainThread())  // 观察线程（主线程）
            .timeout(10, TimeUnit.SECONDS)  // 超时控制
            .retry(3)  // 重试逻辑
            .doOnError { e ->
                log("Recognition failed: ${e.message}")
            }
    }
}
```

**RxJava 优势**：

- 强大的操作符库（map, flatMap, filter, timeout, retry 等）
- 内置背压（Backpressure）处理
- 成熟的生态

**RxJava 劣势**：

- 学习曲线陡峭
- 内存占用较大（每个 Observable 对象）
- 调试困难（堆栈跟踪不清晰）

#### Kotlin Coroutines 架构

```kotlin
// 语音识别流程的 Coroutines 实现（推荐用于 AAOS）
class VoiceRecognitionViewModel(
    private val voiceService: VoiceRecognitionService
) : ViewModel() {
    private val _recognitionResult = MutableStateFlow<RecognitionResult?>(null)
    val recognitionResult = _recognitionResult.asStateFlow()
    
    fun startRecognition() {
        viewModelScope.launch {
            try {
                // 录音（耗时操作）
                val audioBytes = withContext(Dispatchers.IO) {
                    voiceService.recordAudio(duration = 5000)  // 5秒
                }
                
                // 发送到 ASR（网络操作）
                val asrResult = withContext(Dispatchers.IO) {
                    voiceService.sendToASR(audioBytes)
                }
                
                // 更新 UI 状态（主线程）
                _recognitionResult.value = parseASRResult(asrResult)
                
            } catch (e: CancellationException) {
                // Coroutine 被取消
                throw e
            } catch (e: Exception) {
                log("Recognition failed: ${e.message}")
            }
        }
    }
}

// StateFlow 与 LiveData 对比
// StateFlow: 热流，有初始值，自动共享
// LiveData: 冷流，无初始值，生命周期感知
```

**Coroutines 优势**：

- 语法简洁（顺序代码风格）
- 内存占用小（轻量级）
- 无背压问题（天然支持 suspend）
- 更好的异常处理
- AAOS 官方推荐

**Coroutines 劣势**：

- 需要 Kotlin
- 调试需要特殊工具

**AudioRecord 细节**（语音识别场景）：

```kotlin
class AudioRecorder(
    private val sampleRate: Int = 16000,  // 16kHz（语音识别标准）
    private val channelConfig: Int = AudioFormat.CHANNEL_IN_MONO,
    private val audioFormat: Int = AudioFormat.ENCODING_PCM_16BIT
) {
    private val bufferSize = AudioRecord.getMinBufferSize(
        sampleRate, channelConfig, audioFormat
    )
    
    private val audioRecord = AudioRecord(
        MediaRecorder.AudioSource.MIC,  // 输入源
        sampleRate,
        channelConfig,
        audioFormat,
        bufferSize * 2  // 通常分配 2 倍最小值以避免丢帧
    )
    
    fun startRecording(): Flow<ByteArray> = flow {
        audioRecord.startRecording()
        val buffer = ByteArray(bufferSize)
        
        try {
            while (currentCoroutineContext().isActive) {
                val readSize = audioRecord.read(buffer, 0, buffer.size)
                if (readSize > 0) {
                    emit(buffer.copyOf(readSize))
                } else {
                    delay(10)  // 避免忙轮询
                }
            }
        } finally {
            audioRecord.stop()
            audioRecord.release()
        }
    }
}

// 使用
audioRecorder.startRecording()
    .flowOn(Dispatchers.IO)
    .collect { audioFrame ->
        // 发送给 ASR 引擎
        asrEngine.processFrame(audioFrame)
    }
```

```kotlin
class AudioPlayer(
    private val sampleRate: Int = 16000,
    private val channelConfig: Int = AudioFormat.CHANNEL_OUT_MONO
) {
    private val audioTrack = AudioTrack(
        AudioAttributes.Builder()
            .setUsage(AudioAttributes.USAGE_ASSISTANT)  // 语音助手属性
            .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
            .build(),
        AudioFormat.Builder()
            .setSampleRate(sampleRate)
            .setChannelMask(channelConfig)
            .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
            .build(),
        AudioManager.getStreamMinVolume(AudioManager.STREAM_MUSIC) * 2,
        AudioTrack.MODE_STREAM,  // 流模式（逐块写入）
        AudioManager.AUDIO_SESSION_ID_GENERATE
    )
    
    fun playAudio(audioFile: File) {
        audioTrack.play()
        
        val buffer = ByteArray(4096)
        audioFile.inputStream().use { input ->
            while (true) {
                val readSize = input.read(buffer)
                if (readSize < 0) break
                
                audioTrack.write(buffer, 0, readSize, 
                    AudioTrack.WRITE_BLOCKING)  // 阻塞直到写入完成
            }
        }
        
        audioTrack.stop()
    }
}
```

#### 音频焦点管理

```kotlin
class AudioFocusManager(
    private val context: Context,
    private val audioManager: AudioManager
) {
    private val focusRequest = AudioFocusRequest.Builder(
        AudioManager.AUDIOFOCUS_GAIN  // 申请焦点类型
    ).setAudioAttributes(
        AudioAttributes.Builder()
            .setUsage(AudioAttributes.USAGE_ASSISTANT)
            .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
            .build()
    ).setOnAudioFocusChangeListener { focusChange ->
        when (focusChange) {
            AudioManager.AUDIOFOCUS_GAIN -> {
                // 获得焦点，恢复播放
                resumeAudio()
            }
            AudioManager.AUDIOFOCUS_LOSS -> {
                // 失去焦点，停止播放（通常是用户启动其他媒体）
                pauseAudio()
            }
            AudioManager.AUDIOFOCUS_LOSS_TRANSIENT -> {
                // 暂时失去焦点（电话来电）
                pauseAudio()
            }
            AudioManager.AUDIOFOCUS_LOSS_TRANSIENT_CAN_DUCK -> {
                // 暂时失去，但可以降低音量（导航语音）
                reduceVolume()
            }
        }
    }.build()
    
    fun requestAudioFocus() {
        val result = audioManager.requestAudioFocus(focusRequest)
        if (result == AudioManager.AUDIOFOCUS_REQUEST_GRANTED) {
            startPlayback()
        }
    }
    
    fun abandonAudioFocus() {
        audioManager.abandonAudioFocusRequest(focusRequest)
    }
}
```

```kotlin
        val audioRecorder = AudioRecorder()

            .flowOn(Dispatchers.IO)
            .collect { audioFrame ->
                // 传入每一帧音频数据

                    audioFrame,
                    audioFrame.size,
                    isFinal = false
                )

                    emit(ASRResult(
                        text = result.text,
                        confidence = result.confidence,
                        isFinal = false
                    ))
                }
            }
        
        // 获取最终结果
        val finalResult = recognizer.getFinalResult()
        emit(ASRResult(
            text = finalResult.text,
            confidence = finalResult.confidence,
            isFinal = true
        ))
    }
```

#### 意图识别与槽位填充

```
│  Slot "Song": "发如雪"                   │
│  Slot "Action": "播放"                   │
└─────────────────────────────────────────┘
           ↓
```

#### 实现方案

```kotlin
// NLU 引擎接口
interface NLUEngine {
    fun processText(text: String): Intent
    fun setLanguage(language: String)
}

// 基于规则的 NLU（简单场景）
class RuleBasedNLU : NLUEngine {
    
    private val intentPatterns = mapOf(
        Regex("播放|放音乐|来首歌") to "PLAY_MUSIC",
        Regex("停止|暂停") to "PAUSE_MUSIC",
        Regex("上一首|下一首") to "NEXT_MUSIC"
    )
    
    private val artistSlotPattern = Regex("的(.+?)(的歌|的)")
    private val songSlotPattern = Regex("《(.+?)》|「(.+?)」")
    
    override fun processText(text: String): Intent {
        // 匹配意图
        val intent = intentPatterns.entries.find { (pattern, _) ->
            pattern.containsMatchIn(text)
        }?.value ?: "UNKNOWN"
        
        // 提取槽位
        val slots = mutableMapOf<String, String>()
        
        artistSlotPattern.find(text)?.let { match ->
            slots["artist"] = match.groupValues[1]
        }
        songSlotPattern.find(text)?.let { match ->
            slots["song"] = match.groupValues[1].ifEmpty { match.groupValues[2] }
        }
        
        return Intent(intent, slots, confidence = 0.8f)
    }
}

// 基于 BERT 的 NLU（高准确率）
class BERTBasedNLU(private val modelPath: String) : NLUEngine {
    
    private val interpreter = TFLiteInterpreter(modelPath)
    
    override fun processText(text: String): Intent {
        // 1. 文本编码
        val tokens = tokenize(text)
        val inputIds = encodeTokens(tokens)
        
        // 2. 推理
        interpreter.run(inputIds)
        
        // 3. 意图识别
        val intentLogits = interpreter.getOutput("intent_logits")
        val intentIdx = intentLogits.argMax()
        val intentConfidence = intentLogits.softmax()[intentIdx]
        
        // 4. 槽位识别
        val slotLogits = interpreter.getOutput("slot_logits")
        val slots = extractSlots(slotLogits, tokens)
        
        return Intent(
            intent = INTENT_LABELS[intentIdx],
            slots = slots,
            confidence = intentConfidence
        )
    }
}

    val slots: Map<String, String>,
```

```kotlin
interface LLMEngine {
    suspend fun generateResponse(
        userInput: String,
        context: ConversationContext
    ): String
    
    suspend fun streamResponse(
        userInput: String,
        context: ConversationContext
    ): Flow<String>

// 本地 LLM（如 Llama, Qwen）
class LocalLLMEngine(
    private val modelPath: String
) : LLMEngine {
    
    private val llm = LLMRuntime.load(modelPath)
    
    override suspend fun generateResponse(
        userInput: String,
        context: ConversationContext
    ): String = withContext(Dispatchers.IO) {
        
        // 1. 构建 Prompt
        val prompt = buildPrompt(userInput, context)
        
        // 2. 执行推理
        val tokens = llm.tokenize(prompt)
        val outputTokens = mutableListOf<Int>()
        
        var token = tokens.last()
        for (i in 0 until MAX_GEN_LENGTH) {
            // KV 缓存优化（避免重复计算）
            val logits = llm.forward(token, context = null)
            token = sampleToken(logits)  // 采样
            
            if (token == EOS_TOKEN) break
            outputTokens.add(token)
        }
        
        // 3. 解码输出
        return@withContext llm.decode(outputTokens)
    }
    
    override suspend fun streamResponse(
        userInput: String,
        context: ConversationContext
    ): Flow<String> = flow {
        
        val prompt = buildPrompt(userInput, context)
        val tokens = llm.tokenize(prompt)
        var token = tokens.last()
        
        for (i in 0 until MAX_GEN_LENGTH) {
            val logits = llm.forward(token, context = null)
            token = sampleToken(logits)
            
            if (token == EOS_TOKEN) break
            
            val text = llm.decodeToken(token)
            emit(text)
            yield()  // 允许取消
        }
    }.flowOn(Dispatchers.IO)
    
    private fun buildPrompt(
        userInput: String,
        context: ConversationContext
    ): String {
        // 模板化 Prompt
        return """
            |你是一个语音助手。
            |
            |对话历史：
            |${context.history.joinToString("\n") { (role, content) ->
                "$role: $content"
            }}
            |
            |用户: $userInput
            |助手:
        """.trimMargin()
    }
}
```

```kotlin
// 本地 TTS（如 UTTS, FastSpeech2）
class LocalTTSEngine(
    modelPath: String
) : TTSEngine {
    
    private val model = FastSpeech2.load(modelPath)
    
    override suspend fun synthesize(
        text: String,
        speaker: String
    ): AudioData = withContext(Dispatchers.IO) {
        
        // 1. 文本归一化
        val normalized = normalizeText(text)
        
        // 2. 分词 (Tokenization)
        val phonemes = tokenize(normalized)
        
        // 3. 模型推理
        val mel = model.forward(phonemes, speakerId = getSpeakerId(speaker))
        
        // 4. Vocoder（声码器）生成波形
        val waveform = vocoder.generate(mel)
        
        return@withContext AudioData(
            sampleRate = 22050,
            audioData = waveform,
            duration = waveform.size / 22050f
        )
    }
    
    private fun normalizeText(text: String): String {
        return text
            .replace("（", "(")
            .replace("）", ")")
            .replace(Regex("[^\\p{L}\\p{N}\\p{P}\\s]"), "")
    }
    
    private fun tokenize(text: String): List<Int> {
        // 文本 → 音素 → Token ID
        val phonemes = text2phoneme(text)
        return phonemes.map { phoneme2id[it] ?: 0 }
    }
}

class AudioPlayback(context: Context) {

            .setUsage(AudioAttributes.USAGE_ASSISTANT)

        try {
            // 请求音频焦点
            val audioManager = context.getSystemService(Context.AUDIO_SERVICE) 
                as AudioManager

                AudioTrack.WRITE_BLOCKING
            )
            
            // 等待播放完成
            while (audioTrack.playbackHeadPosition < audioData.audioData.size) {
                delay(100)
            }
            
            audioTrack.stop()
            audioManager.abandonAudioFocus(null)
            
        } finally {
            audioTrack.release()
        }
    }
}

    val duration: Float
```

### 4.1 Transformer 基础原理

#### 自注意力机制（Self-Attention）

```
输入序列：["我", "想", "播放", "音乐"]
    ↓
Token Embedding + Positional Encoding
    ↓
        Q (Query)        K (Key)         V (Value)
        ↓                ↓               ↓
    [1024]           [1024]           [1024]
    (from "我")      (from "我")      (from "我")
    
    Attention = softmax(Q·K^T / √d_k) · V
    
    对于 "我"，计算与其他所有词的相关性：
    - "我" → "我": 0.1
    - "我" → "想": 0.3
    - "我" → "播放": 0.4
    - "我" → "音乐": 0.2
    
    加权求和：0.1*v_我 + 0.3*v_想 + 0.4*v_播放 + 0.2*v_音乐
    ↓
    输出：更新后的 embedding
```

#### 多头注意力

```kotlin
// Transformer 注意力的 Kotlin 实现
class MultiHeadAttention(
    val numHeads: Int,
    val d: Int,
    val d_k: Int = d / numHeads
) {
    
    fun forward(
        query: Matrix,  // (seq_len, d)
        key: Matrix,    // (seq_len, d)
        value: Matrix   // (seq_len, d)
    ): Matrix {
        
        // 1. 线性投影
        val Q = query * W_q  // (seq_len, d)
        val K = key * W_k
        val V = value * W_v
        
        // 2. 分头
        val Q_heads = splitHeads(Q, numHeads)  // (numHeads, seq_len, d_k)
        val K_heads = splitHeads(K, numHeads)
        val V_heads = splitHeads(V, numHeads)
        
        // 3. 计算注意力
        val attentionOutputs = mutableListOf<Matrix>()
        for (h in 0 until numHeads) {
            val scores = Q_heads[h] matmul K_heads[h].transpose()  // (seq_len, seq_len)
            val scores_scaled = scores / sqrt(d_k.toFloat())
            val weights = softmax(scores_scaled, axis = -1)  // (seq_len, seq_len)
            val output = weights matmul V_heads[h]  // (seq_len, d_k)
            attentionOutputs.add(output)
        }
        
        // 4. 合并头
        val concat = concatenate(attentionOutputs)  // (seq_len, d)
        
        // 5. 输出投影
        return concat * W_o
    }
}
```

```
Input: "播放" "周杰伦" "的" "发如雪"
  ↓
Embedding Layer
  ↓
┌────────────────────────────────────┐  ─┐
│  Multi-Head Attention              │   │
│  (Self-Attention: 词与词的关联)     │   │ ×N Layers
│  ↓                                  │   │
│  Feed-Forward Network               │   │
│  (Position-wise 全连接)             │   │
└────────────────────────────────────┘  ─┘
  ↓

  ↓

  ↓

                    (next token prediction)
```

### 4.2 大语言模型推理优化

#### 量化（Quantization）

```kotlin
// FP32 → INT8 量化（减少内存占用 75%）
class QuantizationStrategy {
    
    fun quantizeModel(model: FloatArray): ByteArray {
        // 1. 找最大值
        val maxVal = model.maxOrNull() ?: 0f
        val minVal = model.minOrNull() ?: 0f
        
        // 2. 线性映射
        val scale = (maxVal - minVal) / 255f
        val zeroPoint = -minVal / scale
        
        // 3. 转换
        return model.map { value ->
            ((value - minVal) / scale + 0.5f).toByte()
        }.toByteArray()
    }
    
    fun dequantize(quantized: ByteArray, scale: Float, zeroPoint: Float): FloatArray {
        return quantized.map { (it.toInt() - zeroPoint.toInt()) * scale }.toFloatArray()
    }
}

// 在 Android 上应用：
class QuantizedLLMEngine(modelPath: String) {
    
    init {

    }
    
    fun inference(input: FloatArray): FloatArray {
        // 自动处理量化/反量化

    }
```

#### KV 缓存优化

```kotlin
// 标准自回归推理（低效）：
// 推理长度 n：需要 n 次前向传递，每次都计算所有 attention

// 使用 KV 缓存（高效）：
class KVCacheEngine(modelPath: String) {
    
    private val kvCache = mutableListOf<Pair<FloatArray, FloatArray>>()
    
    fun generateTokens(
        inputIds: List<Int>,
        maxLen: Int
    ): List<Int> {
        val output = mutableListOf<Int>()
        
        for (step in 0 until maxLen) {
            if (step == 0) {
                // 首个 token：计算完整注意力
                val hidden = forward(inputIds, kvCache = null)
                val token = sampleToken(hidden)
                output.add(token)
                // 保存 K, V 到缓存
                kvCache.add(Pair(savedK, savedV))
            } else {
                // 后续 token：只处理新增的 token
                val hidden = forward(
                    listOf(output.last()),  // 只处理最后一个 token
                    kvCache = kvCache  // 使用缓存的 K, V
                )
                val token = sampleToken(hidden)
                output.add(token)
                kvCache.add(Pair(savedK, savedV))
            }
            
            if (output.last() == EOT_TOKEN) break
        }
        
        return output
    }
    
    // 推理时间复杂度：
    // 不用缓存：O(n²) → 推理时间 10 秒
    // 用缓存：O(n) → 推理时间 1 秒
}
```

```kotlin
// 对于大模型（>10B 参数），可以分层分布到多个设备
class DistributedLLMEngine(
    private val devices: List<DeviceRuntime>  // GPU, NPU, CPU
) {
    
    // Layer 0-4 → GPU (8B params)
    // Layer 5-9 → NPU (6B params)
    // Layer 10-15 → CPU (4B params)
    
    suspend fun forward(input: FloatArray): FloatArray {
        var hidden = input
        
        // 流式传播
        for (i in 0..15) {
            val device = selectDevice(i)
            hidden = withContext(device.dispatcher) {
                device.runtime.runLayer(i, hidden)
            }
        }
        
        return hidden
    }
}
```
